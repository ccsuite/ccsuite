package ccsuite.webproxy;

import java.io.IOException;

/** An account failed BEFORE anything was sent to the client, so the same request can be tried on another account. */
final class AccountFailure extends RuntimeException {
    final String accountId;

    AccountFailure(String accountId, IOException cause) {
        super(String.valueOf(cause.getMessage()), cause);
        this.accountId = accountId;
    }
}

import ccsuite.webclient.*;

import ccsuite.base.config.CredentialsManager;
import ccsuite.webproxy.health.*;
import ccsuite.base.model.CCAPICredential;
import ccsuite.base.model.CCAPIsCredentials;
import ccsuite.base.model.CircuitBreakerConfig;
import ccsuite.base.model.UsageData;

import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.function.Function;

/**
 * Enhanced CCAPI client with circuit breaker health checking and smart account rotation.
 * Automatically:
 * - Monitors account health and failures
 * - Skips accounts with quota exhaustion
 * - Applies cooldown to failing accounts
 * - Selects best available account based on usage and health
 * - Rotates accounts to distribute load uniformly
 * - Watches credentials file for dynamic updates
 * - Logs account selection and load distribution
 *
 * Usage:
 * 1. Create with circuit breaker config and credentials manager
 * 2. Client automatically refreshes usage and selects best account
 * 3. Use executeWithAutoRotation() for automatic account selection
 * 4. Call startCredentialsFileWatcher() to monitor credentials file changes
 */
public class CCAPIClientWithHealthCheck {

    private final CCAPIClient client;
    private final AccountHealthMonitor healthMonitor;
    private final UsageBasedAccountSelector accountSelector;
    private final AccountRotationLogger logger;
    private final CredentialsManager credentialsManager;
    private CredentialsFileWatcher fileWatcher;

    // Current credentials (updated dynamically)
    private volatile List<CCAPICredential> currentCredentials;

    /**
     * Create client with circuit breaker configuration.
     * Uses default credentials manager.
     */
    public CCAPIClientWithHealthCheck(CircuitBreakerConfig config) {
        this(config, null, new CredentialsManager());
    }

    /**
     * Create client with circuit breaker and request/response file logger.
     * Uses default credentials manager.
     */
    public CCAPIClientWithHealthCheck(CircuitBreakerConfig config,
                                     ccsuite.webproxy.RequestResponseFileLogger fileLogger) {
        this(config, fileLogger, new CredentialsManager());
    }

    /**
     * Create client with circuit breaker, file logger, and custom credentials manager.
     */
    public CCAPIClientWithHealthCheck(CircuitBreakerConfig config,
                                     ccsuite.webproxy.RequestResponseFileLogger fileLogger,
                                     CredentialsManager credentialsManager) {
        this.client = fileLogger != null ? new CCAPIClient(new RequestResponseLoggingInterceptor(fileLogger)) : new CCAPIClient();
        this.healthMonitor = new AccountHealthMonitor(config);
        this.logger = new AccountRotationLogger();
        this.accountSelector = new UsageBasedAccountSelector(client, healthMonitor, logger);
        this.credentialsManager = credentialsManager;
        this.currentCredentials = new ArrayList<>();

        // Load initial credentials
        loadCredentials();
    }

    /**
     * Load credentials from manager
     */
    private void loadCredentials() {
        try {
            var credentials = credentialsManager.load();
            this.currentCredentials = new ArrayList<>(credentials.getActiveCredentials());
            logger.logCredentialsReloaded(currentCredentials.size(), credentials.credentials().size());
        } catch (Exception e) {
            System.err.println("[ROTATION] Failed to load credentials: " + e.getMessage());
            this.currentCredentials = new ArrayList<>();
        }
    }

    /**
     * Handle credentials reload (called by file watcher)
     */
    private void handleCredentialsReload(CCAPIsCredentials newCredentials) {
        var oldCredentials = this.currentCredentials;
        this.currentCredentials = new ArrayList<>(newCredentials.getActiveCredentials());

        // Clear health for removed accounts
        var oldIds = oldCredentials.stream().map(CCAPICredential::id).toList();
        var newIds = currentCredentials.stream().map(CCAPICredential::id).toList();

        for (var oldId : oldIds) {
            if (!newIds.contains(oldId)) {
                healthMonitor.reset(oldId);
            }
        }
    }

    /**
     * Start watching credentials file for changes.
     * When file changes, credentials are automatically reloaded.
     */
    public void startCredentialsFileWatcher() {
        if (fileWatcher != null && fileWatcher.isRunning()) {
            return; // Already running
        }

        fileWatcher = new CredentialsFileWatcher(
            credentialsManager,
            this::handleCredentialsReload,
            logger
        );
        fileWatcher.start();
    }

    /**
     * Stop watching credentials file
     */
    public void stopCredentialsFileWatcher() {
        if (fileWatcher != null) {
            fileWatcher.stop();
            fileWatcher = null;
        }
    }

    /**
     * Execute operation with automatic account selection and rotation.
     * Selects best available account based on usage and health.
     *
     * @param operation Operation to execute with selected credential
     * @param <T> Return type
     * @return Operation result
     * @throws IOException if no accounts available or operation fails
     */
    public <T> T executeWithAutoRotation(Function<CCAPICredential, T> operation) throws IOException {
        // Select best account
        var credential = accountSelector.selectBestAccount(currentCredentials);

        if (credential == null) {
            throw new IOException("No available accounts for request");
        }

        // Execute operation with health tracking
        return executeWithHealthCheck(credential, c -> operation.apply(credential));
    }

    /**
     * Refresh usage data for all accounts and update health status.
     * Call this before selecting accounts to ensure fresh quota information.
     */
    public void refreshUsage(List<CCAPICredential> credentials) {
        for (var credential : credentials) {
            try {
                var usage = ccsuite.webproxy.health.UsageFetcher.fetch(client, credential);
                healthMonitor.updateUsage(credential.id(), usage);
            } catch (IOException e) {
                // Failed to fetch usage - classify and record error
                healthMonitor.recordFailure(credential.id(), e);
            }
        }
    }

    /**
     * Refresh usage for current credentials
     */
    public void refreshUsage() {
        refreshUsage(currentCredentials);
    }

    /**
     * Refresh usage for a single account if stale.
     */
    public void refreshUsageIfStale(CCAPICredential credential) {
        if (healthMonitor.isUsageStale(credential.id())) {
            try {
                var usage = ccsuite.webproxy.health.UsageFetcher.fetch(client, credential);
                healthMonitor.updateUsage(credential.id(), usage);
            } catch (IOException e) {
                healthMonitor.recordFailure(credential.id(), e);
            }
        }
    }

    /**
     * Get best available account from a list.
     * Automatically filters out unavailable accounts and sorts by health/usage.
     * Returns null if no accounts available.
     */
    public CCAPICredential selectBestAccount(List<CCAPICredential> credentials) {
        // Refresh usage for accounts with stale data
        credentials.forEach(this::refreshUsageIfStale);

        return healthMonitor.selectBestAccount(credentials);
    }

    /**
     * Get all available accounts sorted by health and usage.
     */
    public List<CCAPICredential> getAvailableAccounts(List<CCAPICredential> credentials) {
        credentials.forEach(this::refreshUsageIfStale);
        return healthMonitor.getAvailableAccounts(credentials);
    }

    /**
     * Execute an operation with automatic health checking and failure tracking.
     * If operation fails, records failure and applies circuit breaker logic.
     * If operation succeeds, records success and clears failure state.
     *
     * @param credential the credential to use
     * @param operation the operation to execute
     * @param <T> return type
     * @return operation result
     * @throws IOException if operation fails
     */
    public <T> T executeWithHealthCheck(CCAPICredential credential,
                                        Function<CCAPIClient, T> operation) throws IOException {
        // Check if account is available
        if (!healthMonitor.isAccountAvailable(credential.id())) {
            var health = healthMonitor.getHealth(credential.id());
            throw new IOException(String.format(
                "Account %s is not available: %s",
                credential.id(), health.getStatusString()));
        }

        try {
            // Execute operation
            var result = operation.apply(client);

            // Record success
            healthMonitor.recordSuccess(credential.id());

            return result;
        } catch (RuntimeException e) {
            // Unwrap IOException if wrapped in RuntimeException
            if (e.getCause() instanceof IOException ioException) {
                healthMonitor.recordFailure(credential.id(), ioException);
                throw ioException;
            }
            throw e;
        } catch (Exception e) {
            // Handle any other exceptions
            healthMonitor.recordFailure(credential.id(), FailureType.GENERIC_ERROR);
            throw new IOException("Unexpected error: " + e.getMessage(), e);
        }
    }

    /**
     * Fetch usage with health tracking.
     */
    public UsageData fetchUsage(CCAPICredential credential) throws IOException {
        return executeWithHealthCheck(credential, c -> {
            try {
                return c.fetchUsage(credential);
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        });
    }

    /**
     * Send message with health tracking.
     */
    public String sendMessage(CCAPICredential credential, String conversationId, String message)
            throws IOException {
        return sendMessage(credential, conversationId, message, null);
    }

    public String sendMessage(CCAPICredential credential, String conversationId, String message, String model)
            throws IOException {
        return executeWithHealthCheck(credential, c -> {
            try {
                return c.sendMessage(credential, conversationId, message, model);
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        });
    }

    /**
     * Get the underlying CCAPIClient for direct access.
     */
    public CCAPIClient getClient() {
        return client;
    }

    /**
     * Get the health monitor for direct access.
     */
    public AccountHealthMonitor getHealthMonitor() {
        return healthMonitor;
    }

    /**
     * Get health summary for all tracked accounts.
     */
    public String getHealthSummary() {
        return healthMonitor.getHealthSummary();
    }

    /**
     * Check if an account is available.
     */
    public boolean isAccountAvailable(String accountId) {
        return healthMonitor.isAccountAvailable(accountId);
    }

    /**
     * Get current active credentials
     */
    public List<CCAPICredential> getCurrentCredentials() {
        return new ArrayList<>(currentCredentials);
    }

    /**
     * Get account rotation logger
     */
    public AccountRotationLogger getLogger() {
        return logger;
    }

    /**
     * Get account selector
     */
    public UsageBasedAccountSelector getAccountSelector() {
        return accountSelector;
    }

    /**
     * Shutdown the client and stop file watcher.
     */
    public void shutdown() {
        stopCredentialsFileWatcher();
        client.shutdown();
    }
}

/**
 * Anthropic-Compatible API Proxy for CCAPI
 *
 * <p>Implements Anthropic Messages API (/v1/messages) that proxies to CCAPI:
 * <ul>
 *   <li>Converts Anthropic request format to CCAPI prompt format</li>
 *   <li>Creates temporary CCAPI conversations</li>
 *   <li>Streams responses in Anthropic SSE format</li>
 *   <li>Handles both streaming and non-streaming requests</li>
 * </ul>
 *
 * <p>Usage:
 * <pre>
 * // Start proxy on port 8080 with settings from default location
 * var proxy = new CCAPIProxy(8080);
 * proxy.start();
 * </pre>
 *
 * <p>Compatible with Claude Code and other Anthropic API clients.
 */
public class CCAPIProxy {

    private final Javalin app;
    private final CCAPIClient ccapiClient;
    private final CCAPIClientWithHealthCheck healthCheckClient;
    private final List<CCAPICredential> activeCredentials;
    private int currentCredentialIndex = 0;
    private final ObjectMapper objectMapper;
    private final DebugLogger debugLogger;
    private final int port;
    private final boolean individualMessagesVisible;
    private final ccsuite.webproxy.config.Settings.ProxySettings proxySettings;
    /** One web chat per Claude Code session (PRP 74, ccapi.chatReuse=on). */
    private final ChatSessions sessions = ChatSessions.persistent();   // PRP 88: chats are remembered across restarts, without a time limit
    /** The job tag (header x-ccacc-job) of the request this thread is serving, for the call log. */
    private static final ThreadLocal<String> JOB = new ThreadLocal<>();
    /** Request header: the account a new conversation should start on. Response headers: the account that served the call, and the size of the prompt built for it. */
    public static final String PLACE_HEADER = "x-ccacc-place", ACCOUNT_HEADER = "x-ccacc-account", PROMPT_HEADER = "x-ccacc-prompt-chars";
    /** Request header (PRP 86): one line from the host, appended to the prompt built for the web for this call only. Never added to the recorded history or the session key, so chat reuse goes on. */
    public static final String NOTE_HEADER = "x-ccacc-note";

    /** The prompt with the host's note (if any) on its end. */
    static String withNote(String prompt, String note) {
        return note == null || note.isBlank() ? prompt : prompt + "\n\n" + note.strip();
    }
    /** Told the usage gauge each reply of a free account carries (account id, gauge): a free reading from the traffic itself, no probe. */
    public static volatile java.util.function.BiConsumer<String, CCAPIClient.Gauge> gaugeListener;

    /**
     * Create proxy with default settings location
     */
    public CCAPIProxy(int port) throws Exception {
        this(port, (String) null);
    }

    /**
     * Create proxy with specific settings file path
     */
    public CCAPIProxy(int port, String settingsPath) throws Exception {
        this(port, settingsPath, loadAccounts());
    }

    /**
     * Serves from the given accounts instead of reading them from the credentials file (breaker settings still come from that file). The proxy's own
     * selection (health and gauge) does not look at an account's {@code active} flag, so accounts that are inactive in the file are served and failed
     * over like any other; which accounts to pass is the caller's choice.
     */
    public CCAPIProxy(int port, List<CCAPICredential> accounts) throws Exception {
        this(port, (String) null, given(accounts));
    }

    private static Accounts given(List<CCAPICredential> accounts) throws Exception {
        if (accounts == null || accounts.isEmpty()) throw new IllegalArgumentException("No accounts given");
        return new Accounts(List.copyOf(accounts), new CredentialsManager().load().getCircuitBreakerConfig());
    }

    /** The accounts the proxy serves and the breaker settings. */
    record Accounts(List<CCAPICredential> active, ccsuite.base.model.CircuitBreakerConfig breaker) { }

    private static Accounts loadAccounts() throws Exception {
        // Load credentials from shared CCAPIsCredentials.xml
        var credentialsManager = new CredentialsManager();
        var credentials = credentialsManager.load();

        // Get all active credentials for rotation
        var chosenIds = System.getProperty("ccapi.accounts");
        var active = chosenIds == null
            ? credentials.getActiveCredentials()
            : java.util.Arrays.stream(chosenIds.split(",")).map(String::trim).map(credentials::getById).filter(java.util.Objects::nonNull).toList();
        if (active.isEmpty()) {
            throw new IllegalStateException("No active credentials found in CCAPIsCredentials.xml");
        }
        return new Accounts(active, credentials.getCircuitBreakerConfig());
    }

    /** For embedding and tests: the accounts are given instead of read from the credentials file. */
    CCAPIProxy(int port, String settingsPath, Accounts accounts) {
        this.port = port;
        this.activeCredentials = accounts.active();

        System.out.println("[PROXY] Loaded " + activeCredentials.size() + " active credential(s):");
        for (var cred : activeCredentials) {
            System.out.println("  - " + cred.id() + " (" + cred.name() + ")");
        }

        // Load proxy-specific settings from CCAPIProxySettings.xml
        var proxySettingsManager = settingsPath != null
            ? new ProxySettingsManager(settingsPath)
            : new ProxySettingsManager();

        this.proxySettings = proxySettingsManager.load();

        // Chats are temporary (anonymous) unless the settings file asks for visible ones, so the web interface does not fill up
        this.individualMessagesVisible = this.proxySettings != null
            && this.proxySettings.ccapi() != null
            ? this.proxySettings.ccapi().individualMessagesVisible()
            : false;

        // Create JSON mapper with proper configuration
        this.objectMapper = new ObjectMapper();
        this.objectMapper.registerModule(new com.fasterxml.jackson.datatype.jsr310.JavaTimeModule());
        this.objectMapper.configure(
            com.fasterxml.jackson.databind.DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES,
            false
        );

        // Create debug logger with loaded settings
        this.debugLogger = new DebugLogger(this.proxySettings, objectMapper);

        // Create CCAPI client with file logger if individual file logging is enabled
        this.ccapiClient = debugLogger.isIndividualFileLoggingEnabled()
            ? new CCAPIClient(new RequestResponseLoggingInterceptor(debugLogger.getFileLogger()))
            : new CCAPIClient();

        // Create health check client with circuit breaker from credentials config
        var circuitBreakerConfig = accounts.breaker();
        this.healthCheckClient = debugLogger.isIndividualFileLoggingEnabled()
            ? new CCAPIClientWithHealthCheck(circuitBreakerConfig, debugLogger.getFileLogger())
            : new CCAPIClientWithHealthCheck(circuitBreakerConfig);

        System.out.println("[PROXY] Circuit Breaker: " +
            (circuitBreakerConfig.enabled() ? "ENABLED" : "DISABLED") +
            " (threshold: " + circuitBreakerConfig.failureThreshold() + " failures)");

        // Create Javalin app
        this.app = Javalin.create(config -> {
            config.showJavalinBanner = false;
            config.http.asyncTimeout = 300000L; // 5 minute timeout for long responses
            config.http.maxRequestSize = Long.getLong("ccapi.maxRequestBytes", 32L * 1024 * 1024);   // Javalin refuses a body over 1 MB ("Content Too Large", HTTP 500): a long Claude Code conversation is bigger (measured 2026-11-07: 686,000 characters is fine, 1,048,338 was refused before this)
        });

        setupRoutes();
    }

    /**
     * Get the current credential (with health-aware selection and round-robin fallback)
     */
    private synchronized CCAPICredential getCurrentCredential() {
        // Try to get best available account based on health and usage
        var bestAccount = healthCheckClient.selectBestAccount(activeCredentials);

        if (bestAccount != null) {
            debugLogger.logInfo("Selected account: " + bestAccount.id() +
                " (usage: " + getUsagePercent(bestAccount) + "%)");
            return bestAccount;
        }

        // Fallback to round-robin if circuit breaker disabled or no accounts available
        debugLogger.logInfo("No healthy accounts available, using round-robin fallback");
        var credential = activeCredentials.get(currentCredentialIndex);
        currentCredentialIndex = (currentCredentialIndex + 1) % activeCredentials.size();
        return credential;
    }

    /**
     * Get usage percentage for an account (for logging)
     */
    private int getUsagePercent(CCAPICredential credential) {
        var health = healthCheckClient.getHealthMonitor().getHealth(credential.id());
        if (health != null && health.latestUsage() != null && health.latestUsage().fiveHour() != null) {
            return (int) health.latestUsage().fiveHour().utilization();
        }
        return 0;
    }

    /**
     * Setup all HTTP routes
     */
    private void setupRoutes() {
        // Health check
        app.get("/health", this::handleHealth);

        // Circuit breaker health status
        app.get("/health/accounts", this::handleAccountsHealth);

        // Anthropic-compatible Messages API
        app.post("/v1/messages", this::handleMessages);

        // Legacy completion endpoint (minimal implementation)
        app.post("/v1/complete", this::handleComplete);
    }

    /**
     * Health check endpoint
     */
    private void handleHealth(Context ctx) {
        var credentialIds = activeCredentials.stream()
                .map(c -> c.id() + " (" + c.name() + ")")
                .toList();

        ctx.json(Map.of(
            "status", "ok",
            "service", "CCAPI Anthropic-Compatible Proxy",
            "version", "1.0",
            "active_credentials", credentialIds,
            "credential_count", activeCredentials.size(),
            "timestamp", Instant.now().toString()
        ));
    }

    /**
     * Account health status endpoint - shows circuit breaker state
     */
    private void handleAccountsHealth(Context ctx) {
        // Return health summary as plain text
        ctx.contentType("text/plain");
        ctx.result(healthCheckClient.getHealthSummary());
    }

    /** How many requests one account may have in flight (ccapi.perAccountConcurrency, default 2). */
    static final int PER_ACCOUNT = Integer.getInteger("ccapi.perAccountConcurrency", 2);
    private final Map<String, java.util.concurrent.Semaphore> inFlight = new java.util.concurrent.ConcurrentHashMap<>();

    private java.util.concurrent.Semaphore gate(String accountId) {
        return inFlight.computeIfAbsent(accountId, id -> new java.util.concurrent.Semaphore(PER_ACCOUNT, true));
    }

    /** The account that holds the session's chat when it is usable (affinity), else the best untried account that has a free slot; when none has, the fallback (its request then waits for a slot). */
    private synchronized CCAPICredential pickWithCapacity(java.util.Set<String> tried, CCAPICredential fallback, String preferred, boolean holdsChat) {
        // PRP 85: the account that holds the session's chat is waited for when it is busy: moving the session costs the whole conversation again (its cache is lost); only a failure moves it
        if (preferred != null && !tried.contains(preferred) && (holdsChat || gate(preferred).availablePermits() > 0)) {
            var mine = activeCredentials.stream().filter(c -> c.id().equals(preferred)).findFirst();
            if (mine.isPresent() && healthCheckClient.selectBestAccount(List.of(mine.get())) != null) return mine.get();
        }
        var open = activeCredentials.stream().filter(c -> !tried.contains(c.id()) && gate(c.id()).availablePermits() > 0).toList();
        var best = open.isEmpty() ? null : healthCheckClient.selectBestAccount(open);
        return best != null ? best : fallback;
    }

    /** The best available account that has not been tried for this request, or null. */
    private synchronized CCAPICredential nextCredential(java.util.Set<String> tried) {
        var rest = activeCredentials.stream().filter(c -> !tried.contains(c.id())).toList();
        return rest.isEmpty() ? null : healthCheckClient.selectBestAccount(rest);
    }

    private ccsuite.webclient.model.SingleConversation createChatOrFail(CCAPICredential credential, boolean isTemporary) {
        try {
            return ccapiClient.createChat(credential, "Proxy-" + Instant.now().toEpochMilli(), isTemporary);
        } catch (IOException e) {
            healthCheckClient.getHealthMonitor().recordFailure(credential.id(), e);
            throw new AccountFailure(credential.id(), e);
        }
    }

    /** The answer of one web turn with chat reuse on: the reply, the chat it went to and whether it continued the session's chat. */
    record Sent(ccsuite.webclient.CCAPIClient.Reply reply, String conversationId, boolean continued) { }

    /** Pause before the one retry of a refused follow-up (ccapi.followUpRetryMillis). */
    static final long RETRY_PAUSE_MILLIS = Long.getLong("ccapi.followUpRetryMillis", 1500L);

    /**
     * One web turn with chat reuse on. A continuation sends only the new content as the next message of the session's chat (parent = the last reply);
     * when that is refused before any output the chat is forgotten and the same account gets a fresh chat with the full prompt: never a half history.
     * A failure of the fresh chat is the caller's (account failover).
     */
    private Sent reuseTurn(CCAPICredential credential, AnthropicRequest request, ChatSessions.Ticket ticket, String prompt,
                           java.util.function.BooleanSupplier outputStarted, java.util.function.Consumer<String> onText) throws IOException {
        var model = WebModels.of(request.getModel());
        if (ticket.continuation()) {
            // Giving up a chat costs a full-price call (the whole conversation is sent again), so a plain refusal (HTTP 429 "Rate limited", nothing generated)
            // is retried once on the same chat first. A timeout is never repeated: the server may be generating and charging (see FollowUpFailure).
            IOException failure = null;
            for (var attempt = 1; attempt <= 2; attempt++) {
                try {
                    var reply = ccapiClient.sendTurnStreaming(credential, ticket.conversationId(), prompt, model, ticket.parentUuid(), onText);
                    System.out.println("[PROXY] chat reused on " + credential.id() + ", turn " + (ticket.turns() + 1) + ", sent " + prompt.length() + " chars" + (attempt > 1 ? " (second try)" : ""));
                    return new Sent(reply, ticket.conversationId(), true);
                } catch (IOException e) {
                    if (outputStarted.getAsBoolean()) throw e;
                    failure = e;
                    if (attempt == 1 && (FollowUpFailure.of(e) == FollowUpFailure.RETRY_SAME_CHAT || FollowUpFailure.of(e) == FollowUpFailure.NOT_SENT)) {
                        System.out.println("[PROXY] follow-up refused on " + credential.id() + " (" + shorten(e.getMessage()) + "); trying the same chat once more");
                        try { Thread.sleep(RETRY_PAUSE_MILLIS); } catch (InterruptedException ie) { Thread.currentThread().interrupt(); throw e; }
                        continue;
                    }
                    break;
                }
            }
            var what = FollowUpFailure.of(failure);
            if (what == FollowUpFailure.NOT_SENT) { sessions.keep(ticket); throw failure; }   // never sent: the chat is intact, the client may try again on it
            sessions.abandon(ticket);
            if (what == FollowUpFailure.DO_NOT_REPEAT || what == FollowUpFailure.FAIL_OVER) throw failure;   // a timeout may still be generating; a used-up account cannot serve a fresh chat either
            debugLogger.logInfo("Follow-up refused on " + credential.id() + " (" + shorten(failure.getMessage()) + "); starting a fresh chat");
            System.out.println("[PROXY] follow-up refused on " + credential.id() + " (" + shorten(failure.getMessage()) + "); fresh chat");
            ticket.moved("the web refused to continue its chat on account " + credential.id() + ", so it continues in a new chat");   // PRP 85: said in the reply, logged with the session id
            prompt = MessageToPromptConverter.convert(request);
        }
        var chat = ccapiClient.createChat(credential, "Proxy-" + Instant.now().toEpochMilli(), !individualMessagesVisible);
        var reply = ccapiClient.sendTurnStreaming(credential, chat.uuid(), prompt, model, null, onText);
        return new Sent(reply, chat.uuid(), false);
    }

    /** What came of the tool blocks of one answer (PRP 81): the reply that was finally used, whether the chat can still be continued, how often the model was asked again, how many blocks could not be read, how many were repaired. */
    record Settled(Sent sent, boolean broken, int answered, int unparsed, int repaired) { }

    /**
     * PRP 81 R2 and R3. A reply with no call that can run but a block that could not be read is never shown as if it were the answer and never dropped:
     * the error goes back to the model as the next message of its own chat (the parse error with its position, how to escape, or a file for long values) and
     * the answer that follows is read like the first. The same failure is answered at most {@link ToolCallProblems#MAX_ANSWERS} times; then the user is told
     * once, in plain words. Every unreadable and every repaired block goes to the {@link ToolCallLog}.
     *
     * @param streamInto where the follow-up's text goes as it arrives (the streaming answer); null for a non-streaming answer, which is fed from the finished reply
     */
    private Settled settleUnreadable(SseToolBridge bridge, CCAPICredential credential, AnthropicRequest request, ChatSessions.Ticket ticket,
                                     ccsuite.webclient.model.SingleConversation conversation, Sent sent, java.util.function.Consumer<String> streamInto) {
        var model = WebModels.of(request.getModel());
        var session = ticket.sessionId();
        var answered = 0;
        var broken = false;
        while (bridge.needsRetry() && answered < ToolCallProblems.MAX_ANSWERS && !broken) {
            var bad = List.copyOf(bridge.unreadable());
            answered++;
            for (var b : bad) ToolCallLog.record(credential.id(), session, b.raw(), b.reason(), "returned as an error");
            System.out.println("[PROXY] tool call could not be read on " + credential.id() + " (" + shorten(bad.get(0).reason()) + "); asking the model again (" + answered + " of " + ToolCallProblems.MAX_ANSWERS + ")");
            var ask = ToolCallProblems.retryPrompt(bad);
            bridge.restart();
            try {
                String text;
                if (sent != null) {
                    var reply = ccapiClient.sendTurnStreaming(credential, sent.conversationId(), ask, model, sent.reply().uuid(), streamInto);
                    sent = new Sent(reply, sent.conversationId(), sent.continued());
                    text = reply.text();
                } else {
                    text = ccapiClient.sendMessageStreaming(credential, conversation.uuid(), ask, model, streamInto);
                }
                if (streamInto == null) bridge.feed(text);
                bridge.finishRound();
            } catch (IOException e) {
                broken = true;   // what the web chat holds is no longer what the session knows: the next call starts a fresh chat
                debugLogger.logInfo("Asking again about an unreadable tool call failed: " + e.getMessage());
                bridge.unreadable().addAll(bad);
            }
        }
        if (bridge.needsRetry()) {
            var bad = List.copyOf(bridge.unreadable());
            for (var b : bad) ToolCallLog.record(credential.id(), session, b.raw(), b.reason(), "surfaced");
            bridge.say(ToolCallProblems.surface(bad, answered));
        } else if (bridge.hasCalls()) {
            for (var b : bridge.unreadable()) ToolCallLog.record(credential.id(), session, b.raw(), b.reason(), "not run");   // other calls of the same reply ran; this one did not
        }
        for (var r : bridge.repairs()) ToolCallLog.record(credential.id(), session, r.original(), "repaired: " + r.changes(), "repaired");
        bridge.problems().forEach(p -> debugLogger.logInfo("Tool block problem: " + p));
        return new Settled(sent, broken, answered, bridge.unparsed(), bridge.repairs().size());
    }

    /** A call was served: the usage gauge its reply carried goes to the health monitor and the listener, and one line to the call log. */
    private void served(AnthropicRequest request, CCAPICredential credential, ChatSessions.Ticket ticket, Sent sent, int promptChars, int replyChars, int toolCalls, long startNanos, Settled tools) {
        var gauge = sent == null ? null : sent.reply().gauge();
        if (gauge != null && gauge.fiveHour() != null) {
            if (credential.tier() == 0) healthCheckClient.getHealthMonitor().updateUsage(credential.id(), ccsuite.webproxy.health.UsageFetcher.toUsage(new FreeGauge.Reading(gauge.fiveHour(), gauge.sevenDay(), null, gauge.fiveResetsAt(), null, 0, 0)));
            var listener = gaugeListener;
            if (listener != null) listener.accept(credential.id(), gauge);
        }
        var call = callOf(request, credential, ticket, promptChars, startNanos);
        call.put("mode", sent == null ? "legacy" : sent.continued() ? "continued" : ticket.continuation() ? "refused-then-new" : ChatSessions.fresh() ? "fresh" : "new");
        call.put("turn", sent != null && sent.continued() ? ticket.turns() + 1 : 1);
        call.put("replyChars", replyChars);
        if (!(sent != null && sent.continued())) call.put("why", ticket.why());
        if (ticket.moved() != null) {   // PRP 85: a session that had a chat and now has another one, on purpose and told to the user
            call.put("moved", ticket.moved());
            System.out.println("[PROXY] session " + ticket.sessionId() + " moved: " + ticket.moved());
        }
        call.put("toolCalls", toolCalls);
        if (tools != null) {   // PRP 81: tool blocks that could not be read, blocks that were repaired, times the model was asked again
            if (tools.unparsed() > 0) call.put("unparsed", tools.unparsed());
            if (tools.repaired() > 0) call.put("repaired", tools.repaired());
            if (tools.answered() > 0) call.put("retried", tools.answered());
        }
        call.put("ok", true);
        if (gauge != null) { call.put("gauge5h", gauge.fiveHour() == null ? null : Math.round(gauge.fiveHour() * 100)); call.put("gauge7d", gauge.sevenDay() == null ? null : Math.round(gauge.sevenDay() * 100)); }
        CallLog.append(call);
    }

    /** A call failed (before or during the answer): one line to the call log. */
    private void failedCall(AnthropicRequest request, CCAPICredential credential, ChatSessions.Ticket ticket, int promptChars, long startNanos, Throwable failure) {
        var call = callOf(request, credential, ticket, promptChars, startNanos);
        var io = failure instanceof IOException x ? x : failure.getCause() instanceof IOException y ? y : null;
        call.put("mode", ticket.active() ? (ticket.continuation() ? "continued" : "new") : "legacy");
        call.put("why", ticket.why());
        call.put("ok", false);
        call.put("failure", io == null ? "other" : ccsuite.webproxy.health.ErrorClassifier.classify(io).name());
        call.put("error", shorten(failure.getMessage()));
        CallLog.append(call);
    }

    private static java.util.LinkedHashMap<String, Object> callOf(AnthropicRequest request, CCAPICredential credential, ChatSessions.Ticket ticket, int promptChars, long startNanos) {
        var call = new java.util.LinkedHashMap<String, Object>();
        call.put("ts", Instant.now().toString());
        call.put("job", JOB.get());
        call.put("account", credential.id());
        call.put("model", request.getModel());
        call.put("session", ticket.sessionId());
        call.put("promptChars", promptChars);
        call.put("ms", (System.nanoTime() - startNanos) / 1_000_000);
        return call;
    }

    /** The reply went out: the session's chat is remembered, so the next request that extends this one continues it. */
    private void rememberChat(ChatSessions.Ticket ticket, AnthropicRequest request, CCAPICredential credential, Sent sent,
                              ccsuite.webproxy.toolcalls.ToolCallParser.ParseResult parsed) {
        sessions.finish(ticket, request, credential.id(), sent.conversationId(), sent.reply().uuid(),
            ChatSessions.Reply.of(parsed.getTextBeforeToolCalls(), parsed.getToolUses()), sent.continued());
    }

    /** Starts the SSE answer once: headers, message_start and the first content block. */
    private void beginStream(Context ctx, java.io.OutputStream out, boolean[] started, String messageId, String model) {
        if (started[0]) return;
        started[0] = true;
        ctx.contentType("text/event-stream");
        ctx.header("Cache-Control", "no-cache");
        ctx.header("Connection", "keep-alive");
        try {
            writeSSE(out, StreamingChunk.messageStart(messageId, model));
            writeSSE(out, StreamingChunk.contentBlockStart(0));
        } catch (IOException e) {
            throw new RuntimeException("Failed to write stream start", e);
        }
    }

    /** Every account failed before output: say so in the API's own error shape, with what each one said. */
    private void respondAllFailed(Context ctx, java.util.Map<String, String> tried) {
        var used = tried.values().stream().anyMatch(m -> m.contains("exceeded_limit"));
        var detail = new StringBuilder(used ? "Every account is used up or unavailable right now. " : "No account could serve the request. ");
        tried.forEach((id, why) -> detail.append(id).append(": ").append(shorten(why)).append("; "));
        ctx.status(used ? 429 : 502).json(Map.of("type", "error", "error", Map.of("type", used ? "rate_limit_error" : "api_error", "message", detail.toString().trim())));
    }

    private static String shorten(String s) {
        var one = String.valueOf(s).replaceAll("\\s+", " ");
        return one.length() > 110 ? one.substring(0, 110) + "..." : one;
    }

    /**
     * Main /v1/messages endpoint - Anthropic Messages API compatible
     */
    private void handleMessages(Context ctx) {
        CCAPICredential credential = null;
        try {
            // Start new conversation log file
            debugLogger.startNewConversation();
            JOB.set(ctx.header("x-ccacc-job"));

            // Get credential for this request (round-robin rotation)
            credential = getCurrentCredential();
            debugLogger.logInfo("Using credential: " + credential.id() + " (" + credential.name() + ")");

            // Get raw JSON body for logging
            var rawBody = ctx.body();

            // Parse Anthropic request
            var request = objectMapper.readValue(rawBody, AnthropicRequest.class);

            // Log incoming request
            debugLogger.logAnthropicRequest(request, rawBody);

            // Validate request
            if (request.getMessages() == null || request.getMessages().isEmpty()) {
                ctx.status(400).json(Map.of(
                    "error", "invalid_request_error",
                    "message", "messages array is required"
                ));
                return;
            }

            // Check if streaming is requested
            boolean streaming = request.getStream() != null && request.getStream();

            // An account that fails before any output (used up, rate limited, refused, unreachable) is recorded and the request goes to the next best one.
            var tried = new java.util.LinkedHashMap<String, String>();
            var preferred = ChatSessions.enabled() ? sessions.accountOf(request) : null;   // the account that holds this session's chat
            var holdsChat = preferred != null;
            // A host in front of the proxy (ccacc's free lane) may name the account a NEW conversation should start on (header x-ccacc-place: the one with the most predicted room); a session that already has a chat keeps it, and an account that is unknown, busy or unhealthy is ignored by pickWithCapacity.
            if (preferred == null) preferred = ctx.header(PLACE_HEADER);
            while (true) {
                // An account takes only a few requests at once (measured: two work, four at once on one account get HTTP 429); the others wait for a slot,
                // and a request prefers an account that has one free.
                credential = pickWithCapacity(tried.keySet(), credential, preferred, holdsChat);
                ctx.header(ACCOUNT_HEADER, credential.id());   // which account serves this attempt, for the host's ledger; sent with the first byte of the answer, so it is the one that answered
                var slot = gate(credential.id());
                try {
                    if (!slot.tryAcquire(120, java.util.concurrent.TimeUnit.SECONDS)) {
                        throw new AccountFailure(credential.id(), new IOException("busy: " + PER_ACCOUNT + " requests already in flight on this account"));
                    }
                    try {
                        if (streaming) {
                            handleStreamingMessages(ctx, request, credential);
                        } else {
                            handleNonStreamingMessages(ctx, request, credential);
                        }
                        return;
                    } finally {
                        slot.release();
                    }
                } catch (AccountFailure failure) {
                    tried.put(failure.accountId, failure.getMessage());
                    var next = nextCredential(tried.keySet());
                    debugLogger.logInfo("Account " + failure.accountId + " failed before any output (" + failure.getMessage() + "); " + (next == null ? "no other account is available" : "trying " + next.id()));
                    System.out.println("[PROXY] " + failure.accountId + " failed (" + shorten(failure.getMessage()) + "); " + (next == null ? "no other account" : "trying " + next.id()));
                    if (next == null) {
                        respondAllFailed(ctx, tried);
                        return;
                    }
                    credential = next;
                }
            }

            } catch (Exception e) {
            var credId = credential != null ? credential.id() : "UNKNOWN";
            debugLogger.logError("Error handling /v1/messages request [Credential: " + credId + "]", e);
            e.printStackTrace();
            ctx.status(500).json(Map.of(
                "error", "internal_server_error",
                "message", e.getMessage() != null ? e.getMessage() : "An error occurred"
            ));
        }
    }

    /**
     * Handle streaming /v1/messages request
     */
    private void handleStreamingMessages(Context ctx, AnthropicRequest request, CCAPICredential credential) throws IOException {
        // One chat per session (PRP 74): a continuation carries only the new content; otherwise the full prompt as always
        var ticket = sessions.begin(request, credential.id());
        var prompt = ticket.continuation() ? ticket.prompt() : MessageToPromptConverter.convert(request);
        var wholeRequestChars = ticket.continuation() ? MessageToPromptConverter.convert(request).length() : prompt.length();   // PRP 83: what Claude Code sent, for the usage block it reads (not the delta that went to the web)
        prompt = withNote(prompt, ctx.header(NOTE_HEADER));   // PRP 86: after the whole-request count, before the web
        ctx.header(PROMPT_HEADER, String.valueOf(prompt.length()));   // for the host's ledger: what the web is charged for

        // Log converted prompt
        debugLogger.logConvertedPrompt(prompt);

        // Terse account routing info
        var health = healthCheckClient.getHealthMonitor().getHealth(credential.id());
        var usageStr = health.latestUsage() != null && health.latestUsage().fiveHour() != null
            ? String.format("%d%%", (int) health.latestUsage().fiveHour().utilization())
            : "?";
        System.out.println(String.format("\n[PROXY] Streaming via %s [%s, usage:%s], %s, prompt %d chars",
            credential.id(), health.state(), usageStr, ticket.continuation() ? "continuing the chat" : "new chat (" + ticket.why() + ")", prompt.length()));
        var turnStart = System.nanoTime();

        // Create temporary conversation (or visible conversation based on settings)
        var isTemporary = !individualMessagesVisible;

        // Create conversation (doesn't need health check - just metadata); with chat reuse on, the turn creates or continues its own chat
        var conversation = ticket.active() ? null : createChatOrFail(credential, isTemporary);

        // Generate message ID
        var messageId = "msg_" + System.currentTimeMillis();
        var model = request.getModel() != null ? request.getModel() : "claude-3-opus-20240229";

        // The SSE answer starts lazily (headers, message_start, first block) with the first byte of the reply, so that a failure BEFORE it
        // can be retried on another account (AccountFailure) instead of reaching Claude Code as an error.
        var started = new boolean[1];
        var outputStream = ctx.outputStream();

        try {
            // Track output tokens (approximate)
            // output tokens come from the reply characters (TokenEstimate), once the reply is complete

            // Track full response for logging
            var fullResponse = new StringBuilder();
            var bridge = ccsuite.webproxy.toolcalls.JsonToolFormat.enabled()
                ? new ccsuite.webproxy.toolcalls.SseToolBridge(chunk -> { beginStream(ctx, outputStream, started, messageId, model); writeChunk(outputStream, chunk); }, new ccsuite.webproxy.toolcalls.ToolCatalog(request.getTools()))
                : null;

            // PRP-24: Record predicted quota usage BEFORE making request
            int inputTokens = TokenEstimate.tokens(prompt);   // the web is charged for the prompt it received
            int estimatedOutputTokens = 1000; // Conservative estimate for streaming (will be corrected by API sync)
            healthCheckClient.getHealthMonitor().recordPredictedRequest(
                credential.id(),
                inputTokens,
                estimatedOutputTokens
            );

            // Send message with streaming callbacks
            // Note: Streaming doesn't go through healthCheckClient because callbacks need to be immediate
            // Health tracking happens via usage monitoring in healthCheckClient.selectBestAccount()
            var told = new boolean[1];   // PRP 85: a moved session is told in the first words of the reply
            java.util.function.Consumer<String> onText = text -> {
                try {
                    beginStream(ctx, outputStream, started, messageId, model);
                    // Log streaming chunk
                    debugLogger.logStreamingChunk(text);
                    fullResponse.append(text);

                    if (bridge != null && !told[0] && ticket.notice() != null) { told[0] = true; bridge.say(ticket.notice()); }
                    if (bridge != null) {
                        bridge.feed(text);
                    } else {
                        // Send content_block_delta event for each text chunk
                        writeSSE(outputStream, StreamingChunk.contentBlockDelta(0, text));
                        outputStream.flush();
                    }
                } catch (IOException e) {
                    throw new RuntimeException("Failed to write streaming chunk", e);
                }
            };
            Sent sent = null;
            if (ticket.active()) {
                sent = reuseTurn(credential, request, ticket, prompt, () -> started[0], onText);
            } else {
                ccapiClient.sendMessageStreaming(credential, conversation.uuid(), prompt, WebModels.of(request.getModel()), onText);
            }

            beginStream(ctx, outputStream, started, messageId, model);
            if (bridge != null && !told[0] && ticket.notice() != null) { told[0] = true; bridge.say(ticket.notice()); }   // a reply that had no text at all
            healthCheckClient.getHealthMonitor().recordSuccess(credential.id());
            System.out.println("[PROXY] answered in " + (System.nanoTime() - turnStart) / 1_000_000 + " ms");

            // Log full response
            debugLogger.logCcapiResponse(fullResponse.toString());

            // Parse tool calls from the full response
            Settled tools = null;
            if (bridge != null) {
                bridge.finishRound();
                tools = settleUnreadable(bridge, credential, request, ticket, conversation, sent, onText);   // PRP 81: an unreadable call goes back to the model, never to the user as a reply
                sent = tools.sent();
                bridge.finish();
            }
            var parseResult = bridge != null
                ? new ccsuite.webproxy.toolcalls.ToolCallParser.ParseResult(bridge.replyText(), bridge.toolUses())
                : ccsuite.webproxy.toolcalls.ToolCallParser.parse(fullResponse.toString(), debugLogger);

            // Check for tool call failure nudge (PRP-15: diagnostic logging)
            checkForToolCallFailure(request, fullResponse.toString(), parseResult.hasToolCalls());

            if (tools != null && tools.broken()) sessions.abandon(ticket);
            else if (sent != null) rememberChat(ticket, request, credential, sent, parseResult);
            served(request, credential, ticket, sent, prompt.length(), fullResponse.length(), parseResult.getToolUses().size(), turnStart, tools);

            if (bridge == null) {
                // Send content_block_stop event for text
                writeSSE(outputStream, StreamingChunk.contentBlockStop(0));

                // If there are tool calls, send them as additional content blocks
                int toolBlockIndex = 1;
                if (parseResult.hasToolCalls()) {
                    for (var toolUse : parseResult.getToolUses()) {
                        // Send content_block_start for tool_use
                        writeSSE(outputStream, StreamingChunk.toolUseBlockStart(toolBlockIndex, toolUse));
                        // Send content_block_stop for tool_use
                        writeSSE(outputStream, StreamingChunk.contentBlockStop(toolBlockIndex));
                        toolBlockIndex++;
                    }
                }
            }

            // Determine stop reason
            String stopReason = parseResult.hasToolCalls() ? "tool_use" : "end_turn";

            // Send message_delta event with stop reason
            var usage = AnthropicUsage.of(
                RequestTokens.of(request, wholeRequestChars),   // PRP 85: from the content, with weights fitted against count_tokens; the one ratio only as the fallback
                TokenEstimate.tokens(fullResponse)
            );
            writeSSE(outputStream, StreamingChunk.messageDelta(stopReason, usage));

            // Send message_stop event
            writeSSE(outputStream, StreamingChunk.messageStop());

            outputStream.flush();

            // Log the converted response (build for logging)
            var response = parseResult.hasToolCalls()
                ? AnthropicResponse.createWithToolUses(
                    messageId, model,
                    parseResult.getTextBeforeToolCalls(),
                    parseResult.getToolUses(),
                    usage
                )
                : AnthropicResponse.create(messageId, model, fullResponse.toString(), usage);
            debugLogger.logConvertedAnthropicResponse(response);

            // Also log to individual file if file logging is enabled
            if (debugLogger.isIndividualFileLoggingEnabled()) {
                try {
                    var json = objectMapper.writerWithDefaultPrettyPrinter().writeValueAsString(response);
                    var seqStr = String.format("%04d", debugLogger.getFileLogger().getCurrentSequence() - 1);
                    debugLogger.getFileLogger().logConvertedAnthropicResponse(seqStr, json);
                } catch (Exception ex) {
                    System.err.println("[PROXY] Failed to log converted response to file: " + ex.getMessage());
                }
            }

        } catch (Exception e) {
            if (FollowUpFailure.notSent(e) || FollowUpFailure.notSent(e.getCause())) sessions.keep(ticket);   // never sent: the chat is intact
            else sessions.abandon(ticket);   // a failed turn leaves the chat in a state the session does not know: the next call starts a fresh chat
            failedCall(request, credential, ticket, prompt.length(), turnStart, e);
            var cause = e instanceof IOException io ? io : (e.getCause() instanceof IOException io2 ? io2 : null);
            if (cause != null) healthCheckClient.getHealthMonitor().recordFailure(credential.id(), cause);
            if (!started[0] && cause != null && !FollowUpFailure.mayHaveStarted(cause)) throw new AccountFailure(credential.id(), cause);   // nothing sent yet: another account may serve it (not after a timeout: the work may be running and charged)
            beginStream(ctx, outputStream, started, messageId, model);
            // Send error event
            writeSSE(outputStream, StreamingChunk.error("api_error", e.getMessage()));
            outputStream.flush();
        }
    }

    /**
     * Handle non-streaming /v1/messages request
     */
    private void handleNonStreamingMessages(Context ctx, AnthropicRequest request, CCAPICredential credential) throws IOException {
        // One chat per session (PRP 74): a continuation carries only the new content; otherwise the full prompt as always
        var ticket = sessions.begin(request, credential.id());
        var prompt = ticket.continuation() ? ticket.prompt() : MessageToPromptConverter.convert(request);
        var wholeRequestChars = ticket.continuation() ? MessageToPromptConverter.convert(request).length() : prompt.length();   // PRP 83: what Claude Code sent, for the usage block it reads (not the delta that went to the web)
        prompt = withNote(prompt, ctx.header(NOTE_HEADER));   // PRP 86: after the whole-request count, before the web
        ctx.header(PROMPT_HEADER, String.valueOf(prompt.length()));   // for the host's ledger: what the web is charged for

        // Log converted prompt
        debugLogger.logConvertedPrompt(prompt);

        // Terse account routing info
        var health = healthCheckClient.getHealthMonitor().getHealth(credential.id());
        var usageStr = health.latestUsage() != null && health.latestUsage().fiveHour() != null
            ? String.format("%d%%", (int) health.latestUsage().fiveHour().utilization())
            : "?";
        System.out.println(String.format("\n[PROXY] Non-streaming via %s [%s, usage:%s]",
            credential.id(), health.state(), usageStr));

        // Create temporary conversation (or visible conversation based on settings)
        var isTemporary = !individualMessagesVisible;

        // Create conversation (doesn't need health check - just metadata); with chat reuse on, the turn creates or continues its own chat
        var conversation = ticket.active() ? null : createChatOrFail(credential, isTemporary);

        // PRP-24: Record predicted quota usage BEFORE making request
        int inputTokens = TokenEstimate.tokens(prompt);   // the web is charged for the prompt it received
        int estimatedOutputTokens = 1000; // Conservative estimate (will be corrected by API sync)
        healthCheckClient.getHealthMonitor().recordPredictedRequest(
            credential.id(),
            inputTokens,
            estimatedOutputTokens
        );

        // Send message and get response
        var turnStart = System.nanoTime();
        String fullText;
        Sent sent = null;
        if (ticket.active()) {
            try {
                sent = reuseTurn(credential, request, ticket, prompt, () -> false, null);
            } catch (IOException e) {
                if (FollowUpFailure.notSent(e)) sessions.keep(ticket); else sessions.abandon(ticket);
                failedCall(request, credential, ticket, prompt.length(), turnStart, e);
                healthCheckClient.getHealthMonitor().recordFailure(credential.id(), e);
                if (FollowUpFailure.mayHaveStarted(e)) throw e;   // the work may be running and charged: do not repeat it on another account
                throw new AccountFailure(credential.id(), e);   // nothing sent yet: another account may serve it
            }
            healthCheckClient.getHealthMonitor().recordSuccess(credential.id());
            fullText = sent.reply().text();
            debugLogger.logCcapiResponse(fullText);
            if (ticket.notice() != null) fullText = ticket.notice() + fullText;   // PRP 85: a moved session is told in the first words of the reply
        } else {
            // with health check wrapper
            String responseText;
            try {
                responseText = healthCheckClient.sendMessage(credential, conversation.uuid(), prompt, WebModels.of(request.getModel()));
            } catch (IOException e) {
                failedCall(request, credential, ticket, prompt.length(), turnStart, e);
                throw new AccountFailure(credential.id(), e);   // nothing sent yet: another account may serve it (the failure itself is already recorded)
            }

            // Log raw SSE response
            debugLogger.logCcapiResponse(responseText);

            // Parse the SSE response to extract text
            var parser = new ccsuite.webclient.streaming.StreamingEventParser(objectMapper);
            fullText = parser.parse(responseText);
        }

        // Parse tool calls from the response
        Settled tools = null;
        ccsuite.webproxy.toolcalls.ToolCallParser.ParseResult parseResult;
        if (ccsuite.webproxy.toolcalls.JsonToolFormat.enabled()) {
            var bridge = new ccsuite.webproxy.toolcalls.SseToolBridge(chunk -> { }, new ccsuite.webproxy.toolcalls.ToolCatalog(request.getTools()));
            bridge.feed(fullText);
            bridge.finishRound();
            tools = settleUnreadable(bridge, credential, request, ticket, conversation, sent, null);   // PRP 81: an unreadable call goes back to the model, never to the user as a reply
            sent = tools.sent();
            bridge.finish();
            parseResult = new ccsuite.webproxy.toolcalls.ToolCallParser.ParseResult(bridge.replyText(), bridge.toolUses());
            if (tools.unparsed() > 0) fullText = bridge.replyText();   // the reply as it now stands: without the blocks that could not be read, with the words for the user when they did not get through
        } else {
            parseResult = ccsuite.webproxy.toolcalls.ToolCallParser.parse(fullText, debugLogger);
        }

        // Check for tool call failure nudge (PRP-15: diagnostic logging)
        checkForToolCallFailure(request, fullText, parseResult.hasToolCalls());

        if (tools != null && tools.broken()) sessions.abandon(ticket);
        else if (sent != null) rememberChat(ticket, request, credential, sent, parseResult);
        served(request, credential, ticket, sent, prompt.length(), fullText.length(), parseResult.getToolUses().size(), turnStart, tools);

        // Generate response
        var messageId = "msg_" + System.currentTimeMillis();
        var model = request.getModel() != null ? request.getModel() : "claude-3-opus-20240229";

        var usage = AnthropicUsage.of(
            RequestTokens.of(request, wholeRequestChars),   // PRP 85: from the content, with weights fitted against count_tokens; the one ratio only as the fallback
            TokenEstimate.tokens(fullText)
        );

        // Create response with tool calls if present
        var response = parseResult.hasToolCalls()
            ? AnthropicResponse.createWithToolUses(
                messageId, model,
                parseResult.getTextBeforeToolCalls(),
                parseResult.getToolUses(),
                usage
            )
            : AnthropicResponse.create(messageId, model, fullText, usage);

        // Log the converted response
        debugLogger.logConvertedAnthropicResponse(response);

        // Also log to individual file if file logging is enabled
        if (debugLogger.isIndividualFileLoggingEnabled()) {
            try {
                var json = objectMapper.writerWithDefaultPrettyPrinter().writeValueAsString(response);
                var seqStr = String.format("%04d", debugLogger.getFileLogger().getCurrentSequence() - 1);
                debugLogger.getFileLogger().logConvertedAnthropicResponse(seqStr, json);
            } catch (Exception e) {
                System.err.println("[PROXY] Failed to log converted response to file: " + e.getMessage());
            }
        }

        ctx.json(response);
    }

    /**
     * Legacy completion endpoint (basic implementation)
     */
    private void handleComplete(Context ctx) {
        ctx.status(501).json(Map.of(
            "error", "not_implemented",
            "message", "Legacy /v1/complete endpoint is not implemented. Use /v1/messages instead."
        ));
    }

    /**
     * Write SSE event to output stream
     */
    private void writeChunk(java.io.OutputStream out, StreamingChunk chunk) {
        try {
            writeSSE(out, chunk);
            out.flush();
        } catch (IOException e) {
            throw new java.io.UncheckedIOException(e);
        }
    }

    private void writeSSE(java.io.OutputStream out, StreamingChunk chunk) throws IOException {
        var json = objectMapper.writeValueAsString(chunk);
        var event = "event: " + chunk.getType() + "\n" +
                    "data: " + json + "\n\n";
        out.write(event.getBytes(java.nio.charset.StandardCharsets.UTF_8));
    }

    /**
     * Check for tool call failure pattern (PRP-15: diagnostic logging)
     * Uses scoring system on last sentence to detect likely failures:
     * - Last sentence ends with colon: +1 point
     * - Last sentence contains intent phrase (I'll, Let me, etc.): +1 point
     * - Score >= 2: Log warning (likely tool call failure)
     * - Score < 2: Ignore (probably legitimate response)
     *
     * Note: "I'll now" / "Let me now" are stronger indicators (future enhancement)
     */
    private void checkForToolCallFailure(AnthropicRequest request, String responseText, boolean hasToolCalls) {
        // Only check if tools were available in the request
        if (request.getTools() == null || request.getTools().isEmpty()) {
            return;
        }

        // If tool calls were successfully generated, no issue
        if (hasToolCalls) {
            return;
        }

        var trimmedResponse = responseText.trim();

        // Extract last sentence (split by . ! ?)
        var sentences = trimmedResponse.split("[.!?]");
        if (sentences.length == 0) {
            return;
        }

        var lastSentence = sentences[sentences.length - 1].trim();
        if (lastSentence.isEmpty()) {
            return;
        }

        var lowerLastSentence = lastSentence.toLowerCase();
        int score = 0;
        var reasons = new java.util.ArrayList<String>();

        // Pattern 1: Last sentence ends with colon (with optional whitespace)
        // Matches: ":" or ": " or " :" or " : "
        if (lastSentence.matches(".*\\s?:\\s?$")) {
            score++;
            reasons.add("ends with colon");
        }

        // Pattern 2: Last sentence contains intent indicators (case-insensitive)
        // Matches: "I'll", "Let me", "I will", "I'm going to", etc.
        if (lowerLastSentence.matches(".*(i'll|let me|i will|i'm going to|i am going to).*")) {
            score++;
            reasons.add("contains intent phrase");
        }

        // Log warning only if score >= 2 (likely failure)
        if (score >= 2) {
            debugLogger.logWarning("[PRP-15] Tool call failure detected! " +
                "Score: " + score + "/3, " +
                "Reasons: " + String.join(", ", reasons) + ". " +
                "Model: " + request.getModel() + ", " +
                "Response length: " + trimmedResponse.length() + " chars, " +
                "Last sentence: \"" + lastSentence + "\"");
        }
    }

                port = Integer.parseInt(args[i + 1]);
                i++;
            } else if (args[i].equals("--settings") && i + 1 < args.length) {
                settingsPath = args[i + 1];
                i++;
            } else if (args[i].equals("--no-tray")) {
                noTray = true;
            } else if (args[i].equals("--help")) {
                printHelp();
                return;
            }
        }

        try {
            var proxy = new CCAPIProxy(port, settingsPath);
            proxy.start();

            // Create system tray icon (unless disabled)
            SystemTrayManager trayManager = null;
            if (!noTray && SystemTrayManager.isSupported()) {
                trayManager = new SystemTrayManager(port);
                final var finalTrayManager = trayManager;
                trayManager.show(() -> {
                    System.out.println("\nShutting down proxy server (via system tray)...");
                    finalTrayManager.hide();
                    proxy.stop();
                    System.out.println("Server stopped.");
                    System.exit(0);
                });
            } else if (!noTray) {
                System.out.println("[PROXY] System tray not available (use Ctrl+C to stop server)");
            }

            // Add shutdown hook for graceful shutdown (Ctrl+C, etc.)
            final var finalTrayManagerForHook = trayManager;
            Runtime.getRuntime().addShutdownHook(new Thread(() -> {
                System.out.println("\nShutting down proxy server...");
                if (finalTrayManagerForHook != null) {
                    finalTrayManagerForHook.hide();
                }
                proxy.stop();
                System.out.println("Server stopped.");

                // Force exit if threads don't die within 2 seconds
                // This prevents hanging when Jetty threads don't shutdown gracefully
                var forceExitThread = new Thread(() -> {
                    try {
                        Thread.sleep(2000);
                        System.err.println("Forcing JVM exit (some threads didn't stop gracefully)...");
                        System.exit(0);
                    } catch (InterruptedException e) {
                        // Shutdown completed before timeout, ignore
                    }
                }, "ForceExitWatchdog");
                forceExitThread.setDaemon(true);
                forceExitThread.start();
            }));

        } catch (Exception e) {
            System.err.println("Failed to start proxy: " + e.getMessage());
            e.printStackTrace();
            System.exit(1);
        }
    }

        } catch (IOException e) {
            if (!warned) { warned = true; System.err.println("[PROXY] call log not written: " + e.getMessage()); }
        }
    }

import module java.base;
import ccsuite.webproxy.anthropic.AnthropicRequest;

import com.fasterxml.jackson.databind.ObjectMapper;

/**
 * One web chat per Claude Code session (PRP 74). A session is recognised by the system prompt, the first user message and the model; a later
 * request continues the chat only when its history is the previous history plus exactly one assistant turn (the one this proxy returned) and
 * one new user message, on the same account with the same tools. Anything else starts a fresh chat: never a half history.
 *
 * <p>Setting {@code ccapi.chatReuse=on|off|fresh}: on by default since 2026-11-06 (measured on real Claude Code sessions, PRP 74 results).
 */
public final class ChatSessions {
    /** A reply that tells the user a session was moved to a new chat (PRP 85) starts with this. */
    public static final String NOTICE_PREFIX = "(fclaude: ";

    public static boolean enabled() {
        return ("on".equalsIgnoreCase(mode()) || fresh()) && JsonToolFormat.enabled();
    }

    /** {@code ccapi.chatReuse}: on (default) | off | fresh (the Messages-style stream, which carries the usage gauge, but every call a new chat and nothing remembered: the control for measuring reuse). */
    static String mode() {
        return System.getProperty("ccapi.chatReuse", "on");
    }

    static boolean fresh() {
        return "fresh".equalsIgnoreCase(mode());
    }

    /** What the proxy returned for the last call: the text and the tool calls (id and name) as Claude Code will echo them back. */
    public record Reply(String text, List<String> toolCalls) {
        public static Reply of(String text, List<ToolUse> uses) {
            return new Reply(text == null ? "" : text.strip(), uses == null ? List.of() : uses.stream().map(u -> u.getId() + "|" + u.getName()).toList());
        }
    }

    /** The state of one session between two calls. */
    static final class Session {
        final String key, accountId, toolsHash, conversationId, parentUuid;
        final List<String> history;
        final Reply reply;
        final int turns;
        volatile boolean busy;
        volatile long used = System.nanoTime();

        Session(String key, String accountId, String toolsHash, String conversationId, String parentUuid, List<String> history, Reply reply, int turns) {
            this.key = key; this.accountId = accountId; this.toolsHash = toolsHash; this.conversationId = conversationId;
            this.parentUuid = parentUuid; this.history = history; this.reply = reply; this.turns = turns;
        }
    }

    /**
     * What the proxy does with one request. {@link #active()} false = reuse is not in play (legacy path, nothing recorded). A continuation carries the
     * chat, the parent message and the prompt made of the new content only; otherwise a new chat is started and (when {@code record}) remembered.
     */
    public static final class Ticket {
        static final Ticket NONE = new Ticket(false, null, null, null, null, null, false, "reuse is off");
        final boolean active;
        final Session session;
        final String key, conversationId, parentUuid, prompt;
        final boolean record;

        /** Why this call starts a new chat (or "continued"): the first call of a session, the history that did not extend, the account that changed ... */
        final String why;

        /** Set when a session that had a chat is moved to a new one (PRP 85): the sentence for the log and for the user; null for a first call and a continuation. */
        volatile String moved;

        Ticket(boolean active, Session session, String key, String conversationId, String parentUuid, String prompt, boolean record, String why) {
            this.why = why;
            this.active = active; this.session = session; this.key = key; this.conversationId = conversationId; this.parentUuid = parentUuid;
            this.prompt = prompt; this.record = record;
        }

        public boolean active() { return active; }
        public boolean continuation() { return session != null; }
        public String why() { return why; }
        public String moved() { return moved; }
        /** Marks this call as a move of its session (a refused follow-up, a lost chat ...) and returns the ticket. */
        public Ticket moved(String what) { moved = what; return this; }
        /** The short sentence the reply starts with when the session was moved (PRP 85), or null. */
        public String notice() { return moved == null ? null : NOTICE_PREFIX + moved + "; this one reply costs more.)\n\n"; }
        public String conversationId() { return conversationId; }
        public String parentUuid() { return parentUuid; }
        public String prompt() { return prompt; }
        public int turns() { return session == null ? 0 : session.turns; }
        /** Short id of the session (for logs); "-" when reuse is not in play. */
        public String sessionId() { return key == null ? "-" : key.substring(0, 8); }
    }

    private final Map<String, Session> sessions = new LinkedHashMap<>();
    /** Sessions whose chat was dropped (a failed turn, a refusal, a used-up account), with the account it was on: so the next call can say why it starts a new chat. */
    private final Map<String, String> lostOn = new LinkedHashMap<>() {
        @Override protected boolean removeEldestEntry(Map.Entry<String, String> e) { return size() > 200; }
    };
    private final int capacity;
    private final long ttlNanos;
    /** PRP 88: where each session's chat is kept across restarts (one small file per session); null keeps them in memory only. */
    private final Path dir;

    /** In memory only, kept until the capacity pushes the least recently used out (no time limit unless {@code ccapi.chatReuse.ttlMinutes} sets one). */
    public ChatSessions() {
        this(Integer.getInteger("ccapi.chatReuse.sessions", 2000), Duration.ofMinutes(Long.getLong("ccapi.chatReuse.ttlMinutes", 0)), null);
    }

    public ChatSessions(int capacity, Duration ttl) { this(capacity, ttl, null); }

    /**
     * The proxy's own store (PRP 88, user 2026-11-09: remember the chat a session worked in indefinitely and try to continue there). A session is written to
     * {@code ccapi.chatReuse.dir} (ccacc sets ~/ccsuite/webproxy/chats; unset means memory only) after every answered call and read back when a request of it arrives, also after a restart;
     * it is forgotten only when its chat is known to be lost. If the web refuses the old chat, the call starts a new one as before, so a miss costs no more than today.
     */
    public static ChatSessions persistent() {
        var d = System.getProperty("ccapi.chatReuse.dir");   // set by the host (ccacc's free lane); unset (tests, a bare proxy) keeps the sessions in memory only
        return new ChatSessions(Integer.getInteger("ccapi.chatReuse.sessions", 2000), Duration.ofMinutes(Long.getLong("ccapi.chatReuse.ttlMinutes", 0)),
                d == null || d.isBlank() ? null : Path.of(d));
    }

    ChatSessions(int capacity, Duration ttl, Path dir) {
        this.capacity = capacity;
        this.ttlNanos = ttl.toNanos();
        this.dir = dir;
    }

    /** The account that holds this request's chat, when the request continues it cleanly (so the router can keep affinity); else null. */
    public synchronized String accountOf(AnthropicRequest request) {
        if (fresh()) return null;
        var s = lookup(keyOf(request));
        return s != null && !s.busy && extendsCleanly(s, request) ? s.accountId : null;
    }

    /** Decides how to serve this request on {@code accountId}; a continuation is claimed (busy) until {@link #finish} or {@link #abandon}. */
    public synchronized Ticket begin(AnthropicRequest request, String accountId) {
        if (!enabled() || request.getTools() == null || request.getTools().isEmpty() || request.getMessages() == null) return Ticket.NONE;
        evict();
        var key = keyOf(request);
        if (fresh()) return new Ticket(true, null, key, null, null, null, false, "control mode: every call a new chat");
        var s = lookup(key);
        if (s != null && s.busy) return new Ticket(true, null, key, null, null, null, false, "another call of the session is in flight");   // another call of the session is in flight: a separate chat, not remembered
        String why = s == null ? (hasHistory(request) ? "no chat held for this session" : "first call of the session") : !s.accountId.equals(accountId) ? "account changed (" + s.accountId + " to " + accountId + ")" : whyNot(s, request);
        if (why == null) {
            var prompt = MessageToPromptConverter.convertContinuation(request, s.history.size() + 1);
            if (!prompt.isBlank()) {
                s.busy = true;
                s.used = System.nanoTime();
                return new Ticket(true, s, key, s.conversationId, s.parentUuid, prompt, true, "continued");
            }
            why = "the new message has no content";
        }
        return new Ticket(true, null, key, null, null, null, true, why).moved(moveOf(s, key, accountId, request));
    }

    static boolean hasHistory(AnthropicRequest request) {
        return request.getMessages().stream().anyMatch(m -> "assistant".equals(m.getRole()));
    }

    /** The sentence for a session that had a chat and now starts a new one (PRP 85); null for the first call of a session. */
    private String moveOf(Session s, String key, String accountId, AnthropicRequest request) {
        var lost = lostOn.remove(key);
        if (s == null) {
            if (!hasHistory(request)) return null;   // nothing was said yet: the first call
            return lost == null ? "the proxy held no chat for this conversation (restarted, or the chat expired), so it continues in a new chat"
                : lost.equals(accountId) ? "the last turn of its chat failed, so it continues in a new chat"
                : "account " + lost + " could not serve it, so it moved to account " + accountId + " in a new chat";
        }
        return !s.accountId.equals(accountId) ? "it moved from account " + s.accountId + " to account " + accountId + " in a new chat"
            : "the conversation was edited or compacted, so it continues in a new chat";
    }

    /** Remembers the chat after a call that was answered: the next request is expected to extend this request by this reply. */
    public synchronized void finish(Ticket ticket, AnthropicRequest request, String accountId, String conversationId, String assistantUuid, Reply reply, boolean continued) {
        if (!ticket.active || !ticket.record) { release(ticket); return; }
        if (assistantUuid == null || assistantUuid.isBlank()) { drop(ticket); return; }
        var turns = continued ? ticket.turns() + 1 : 1;
        sessions.remove(ticket.key);
        lostOn.remove(ticket.key);
        var s = new Session(ticket.key, accountId, toolsHash(request), conversationId, assistantUuid, fingerprints(request.getMessages()), reply, turns);
        sessions.put(ticket.key, s);
        save(s);
        evict();
    }

    /** The call failed or the follow-up was refused: forget the chat of this session, so the next call starts a fresh one. */
    /** The call never reached the web (connect failure): the chat is intact, so the session stays and the next call may continue it. */
    public synchronized void keep(Ticket ticket) {
        release(ticket);
    }

    public synchronized void abandon(Ticket ticket) {
        drop(ticket);
    }

    private void drop(Ticket ticket) {
        if (ticket.session != null && sessions.remove(ticket.key, ticket.session)) { lostOn.put(ticket.key, ticket.session.accountId); forget(ticket.key); }
    }

    private void release(Ticket ticket) {
        if (ticket.session != null) ticket.session.busy = false;
    }

    public synchronized int size() {
        return sessions.size();
    }

    private void evict() {
        var now = System.nanoTime();
        sessions.values().removeIf(s -> now - s.used > (s.busy ? Duration.ofMinutes(15).toNanos() : ttlNanos > 0 ? ttlNanos : Long.MAX_VALUE));   // a claim that was never released (a call that died) does not block the session for good
        while (sessions.size() > capacity) {
            var oldest = sessions.values().stream().filter(s -> !s.busy).min(Comparator.comparingLong(s -> s.used));
            if (oldest.isEmpty()) break;
            sessions.remove(oldest.get().key);
        }
    }

    /**
     * The extension test: same tools; the previous messages unchanged (as the model saw them: cache_control and thinking do not count); then
     * exactly one assistant message that is what this proxy returned, then exactly one user message with something in it.
     */
    static boolean extendsCleanly(Session s, AnthropicRequest request) {
        return whyNot(s, request) == null;
    }

    /** Null when the request extends the session's history cleanly; else the first reason it does not (for the call log). */
    static String whyNot(Session s, AnthropicRequest request) {
        var messages = request.getMessages();
        var n = s.history.size();
        if (messages == null) return "no messages";
        if (messages.size() < n + 2) return "message count " + messages.size() + ", expected at least " + (n + 2);
        if (!s.toolsHash.equals(toolsHash(request))) return "tools changed";
        for (var i = 0; i < n; i++) {
            if (!s.history.get(i).equals(fingerprint(messages.get(i)))) return "message " + i + " (" + messages.get(i).getRole() + ") differs";
        }
        var assistant = messages.get(n);
        if (!"assistant".equals(assistant.getRole())) return "message " + n + " is " + assistant.getRole() + ", expected the assistant turn";
        // Claude Code follows the prompt with attachments (hook output, reminders) as further user messages, so one or more user messages may follow the assistant turn
        for (var i = n + 1; i < messages.size(); i++) {
            var role = messages.get(i).getRole();
            if (!"user".equals(role) && !"system".equals(role)) return "message " + i + " is " + role + ", expected a user or system message";
        }
        var reply = replyOf(assistant);
        if (!reply.text().equals(s.reply.text())) return "assistant text differs at char " + firstDifference(s.reply.text(), reply.text()) + " (" + s.reply.text().length() + " chars returned, " + reply.text().length() + " sent back)";
        if (!reply.toolCalls().equals(s.reply.toolCalls())) return "assistant tool calls differ (returned " + s.reply.toolCalls() + ", sent back " + reply.toolCalls() + ")";
        if (MessageToPromptConverter.convertContinuation(request, n + 1).isBlank()) return "the new messages have no content";
        return null;
    }

    static int firstDifference(String a, String b) {
        var n = Math.min(a.length(), b.length());
        for (var i = 0; i < n; i++) if (a.charAt(i) != b.charAt(i)) return i;
        return n;
    }

    /** The assistant message as the next request carries it, in the same shape as {@link Reply#of}. */
    static Reply replyOf(Message assistant) {
        var content = assistant.getContent();
        if (content == null) return new Reply("", List.of());
        if (content.isTextual()) return new Reply(content.asText().strip(), List.of());
        var text = new StringBuilder();
        var calls = new ArrayList<String>();
        for (var block : content) {
            switch (block.path("type").asText()) {
                case "text" -> text.append(block.path("text").asText());
                case "tool_use" -> calls.add(block.path("id").asText() + "|" + block.path("name").asText());
                default -> { }
            }
        }
        return new Reply(text.toString().strip(), calls);
    }

    /** The key of a session: Claude Code's own session id (when the request carries one), the system prompt, the first user message and the model. Two sessions that open with the same words are two sessions. */
    static String keyOf(AnthropicRequest request) {
        var first = request.getMessages() == null || request.getMessages().isEmpty() ? "" : canonical(request.getMessages().get(0));
        var metadata = request.getMetadata() instanceof Map<?, ?> m && m.get("user_id") instanceof String u ? u : "";
        return hash(sessionIdOf(metadata) + "\u0000" + String.valueOf(request.getSystemPrompt()) + "\u0000" + first + "\u0000" + request.getModel());
    }

    private static final ObjectMapper JSON = new ObjectMapper();

    /** Claude Code's session id out of the request's metadata.user_id (a JSON text with session_id, or the older user_..._session_ID text); "" when there is none. */
    public static String sessionIdOf(String userId) {
        if (userId == null || userId.isBlank()) return "";
        try {
            var id = JSON.readTree(userId).path("session_id").asText("");
            if (!id.isEmpty()) return id;
        } catch (Exception e) { /* not JSON: the older shape */ }
        var at = userId.lastIndexOf("_session_");
        return at < 0 ? "" : userId.substring(at + 9);
    }

    static String toolsHash(AnthropicRequest request) {
        return hash(String.valueOf(request.getTools()));
    }

    static List<String> canonical(List<Message> messages) {
        return messages.stream().map(ChatSessions::canonical).toList();
    }

/** A message as the history keeps it: a short hash of its canonical text (enough to tell it is unchanged, and a file of hashes stays small). */
static String fingerprint(Message message) { return hash(canonical(message)).substring(0, 20); }

static List<String> fingerprints(List<Message> messages) { return messages.stream().map(ChatSessions::fingerprint).toList(); }

static String canonical(Message message) {
        return message.getRole() + "\u0000" + JsonToolFormat.render(message);
    }

    static String hash(String text) {
        try {
            return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(text.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }
    // ---- PRP 88: the sessions on disk

    /** The session in memory, else the one saved on disk (after a restart, or pushed out of memory); null when neither has it. */
    private Session lookup(String key) {
        var s = sessions.get(key);
        if (s != null || dir == null) return s;
        s = load(key);
        if (s != null) { sessions.put(key, s); evict(); }
        return s;
    }

    private Path file(String key) { return dir.resolve(key + ".json"); }

    private void save(Session s) {
        if (dir == null) return;
        try {
            Files.createDirectories(dir);
            var o = JSON.createObjectNode().put("key", s.key).put("accountId", s.accountId).put("toolsHash", s.toolsHash).put("conversationId", s.conversationId)
                    .put("parentUuid", s.parentUuid).put("turns", s.turns).put("saved", Instant.now().toString());
            var h = o.putArray("history");
            s.history.forEach(h::add);
            var r = o.putObject("reply").put("text", s.reply.text());
            var c = r.putArray("toolCalls");
            s.reply.toolCalls().forEach(c::add);
            var tmp = dir.resolve(s.key + ".tmp");
            Files.writeString(tmp, JSON.writeValueAsString(o));
            Files.move(tmp, file(s.key), StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE);
        } catch (IOException | RuntimeException e) {
            System.err.println("[PROXY] chat session not saved: " + e.getMessage());   // fail open: the session still works from memory
        }
    }

    private Session load(String key) {
        var f = file(key);
        if (!Files.isRegularFile(f)) return null;
        try {
            var o = JSON.readTree(f.toFile());
            var history = new ArrayList<String>();
            o.path("history").forEach(n -> history.add(n.asText()));
            var calls = new ArrayList<String>();
            o.path("reply").path("toolCalls").forEach(n -> calls.add(n.asText()));
            return new Session(key, o.path("accountId").asText(), o.path("toolsHash").asText(), o.path("conversationId").asText(), o.path("parentUuid").asText(),
                    List.copyOf(history), new Reply(o.path("reply").path("text").asText(""), List.copyOf(calls)), o.path("turns").asInt(1));
        } catch (IOException | RuntimeException e) {
            return null;   // a torn file: the session starts a new chat, as before
        }
    }

    private void forget(String key) {
        if (dir == null) return;
        try { Files.deleteIfExists(file(key)); } catch (IOException e) { /* a stale file only costs one refused continuation */ }
    }
}

 *   <li>File output with timestamps (one file per conversation)</li>
 *   <li>Pretty-printed JSON</li>

 * </ul>

public class DebugLogger {

    private final Settings.ProxySettings.ConversationDebug debugSettings;
    private final ObjectMapper objectMapper;
    private final Path conversationLogsDir;
    private Path currentConversationFile;
    private final RequestResponseFileLogger fileLogger;

    private static final DateTimeFormatter TIMESTAMP_DISPLAY =
        DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss.SSS")
                         .withZone(ZoneId.of("UTC"));

    private static final DateTimeFormatter FILE_NAME_FORMAT =
        DateTimeFormatter.ofPattern("yyyy-MM-dd_HHmm")
                         .withZone(ZoneId.of("UTC"));

    private static final DateTimeFormatter RUN_FOLDER_FORMAT =
        DateTimeFormatter.ofPattern("yyyy-MM-dd_HHmmss")
                         .withZone(ZoneId.of("UTC"));

    /**
     * Create debug logger with settings
     */
    public DebugLogger(Settings.ProxySettings proxySettings, ObjectMapper objectMapper) {
        this.debugSettings = proxySettings != null && proxySettings.conversationDebug() != null
            ? proxySettings.conversationDebug()
            : null;
        this.objectMapper = objectMapper;

        // Set up conversation logs directory with timestamped folder per server run
        // <userhome>\xyz-jphil\ccapis\conversations-logs\yyyy-MM-dd_HHmmss\
        var userHome = System.getProperty("user.home");
        var baseLogsDir = Path.of(userHome, "xyz-jphil", "ccapis", "conversations-logs");
        var runTimestamp = RUN_FOLDER_FORMAT.format(Instant.now());
        this.conversationLogsDir = baseLogsDir.resolve(runTimestamp);

        // Create directory for this run
        try {
            Files.createDirectories(conversationLogsDir);
            System.out.println("[DEBUG] Conversation logs directory: " + conversationLogsDir);
        } catch (IOException e) {
            System.err.println("[DEBUG] Failed to create conversations-logs directory: " + e.getMessage());
        }

        // Clear old run folders if configured (keeps only current run)
        if (isFileOutputEnabled() && debugSettings.outputToFile().clearPreviousAtStart()) {
            try {
                if (Files.exists(baseLogsDir)) {
                    Files.list(baseLogsDir)
                         .filter(Files::isDirectory)
                         .filter(p -> !p.equals(conversationLogsDir)) // Don't delete current run
                         .forEach(p -> {
                             try {
                                 // Delete the entire old run folder
                                 Files.walk(p)
                                      .sorted((a, b) -> b.compareTo(a)) // Delete files before dirs
                                      .forEach(path -> {
                                          try {
                                              Files.delete(path);
                                          } catch (IOException e) {
                                              System.err.println("[DEBUG] Failed to delete: " + path);
                                          }
                                      });
                             } catch (IOException e) {
                                 System.err.println("[DEBUG] Failed to delete run folder: " + p);
                             }
                         });
                }
            } catch (IOException e) {
                System.err.println("[DEBUG] Failed to clear old run folders: " + e.getMessage());
            }
        }

        // Initialize individual file logger if enabled
        if (isFileOutputEnabled() && debugSettings.outputToFile().saveAsIndividualFiles()) {
            var createIndex = debugSettings.outputToFile().createIndexFile();
            this.fileLogger = new RequestResponseFileLogger(conversationLogsDir, createIndex);
            System.out.println("[DEBUG] Individual file logging enabled with index=" + createIndex);
        } else {
            this.fileLogger = null;
        }
    }

    /**
     * Start a new conversation log file

                     "CONVERSATION LOG - Started at " + TIMESTAMP_DISPLAY.format(timestamp) + " UTC\n" +
                     "=".repeat(80) + "\n\n";
        logToFile(header);
    }

    /**
     * Log incoming Anthropic request
     */
    public void logAnthropicRequest(AnthropicRequest request, String rawJson) {
        if (!isInputLoggingEnabled()) {
            return;
        }

        var sb = new StringBuilder();
        sb.append("\n").append("=".repeat(80)).append("\n");
        sb.append("[").append(timestamp()).append("] INCOMING ANTHROPIC REQUEST\n");
        sb.append("=".repeat(80)).append("\n");

        // Log model
        sb.append("Model: ").append(request.getModel()).append("\n");
        sb.append("Stream: ").append(request.getStream()).append("\n");
        sb.append("Max Tokens: ").append(request.getMaxTokens()).append("\n\n");

        // Log system prompt
        var systemPrompt = request.getSystemPrompt();
        if (systemPrompt != null && !systemPrompt.isEmpty()) {
            sb.append("--- SYSTEM PROMPT ---\n");
            sb.append(truncateForConsole(systemPrompt)).append("\n");
            sb.append("System prompt length: ").append(systemPrompt.length()).append(" chars\n\n");
        }

        // Log messages
        sb.append("--- MESSAGES (").append(request.getMessages().size()).append(") ---\n");
        for (int i = 0; i < request.getMessages().size(); i++) {
            var msg = request.getMessages().get(i);
            var content = msg.getTextContent();
            sb.append("\n[").append(i).append("] Role: ").append(msg.getRole()).append("\n");
            sb.append(truncateForConsole(content)).append("\n");
            sb.append("Content length: ").append(content.length()).append(" chars\n");
        }

        // Log raw JSON
        sb.append("\n--- RAW JSON ---\n");
        try {
            var prettyJson = objectMapper.writerWithDefaultPrettyPrinter()
                                        .writeValueAsString(request);
            sb.append(truncateForConsole(prettyJson)).append("\n");
        } catch (Exception e) {
            sb.append(truncateForConsole(rawJson)).append("\n");
        }

        sb.append("=".repeat(80)).append("\n");

        logToConsole(sb.toString());
        logToFile(sb.toString());
    }

    /**
     * Truncate text for console output based on stdOutLimit
     */
    private String truncateForConsole(String text) {
        if (text == null) {
            return "(null)";
        }

        int limit = debugSettings != null ? debugSettings.stdOutLimit() : 1000;

        if (text.length() <= limit) {
            return text;
        }

        return text.substring(0, limit) + "\n... (truncated " + (text.length() - limit) + " chars)";
    }

        System.out.flush();

            );
        } catch (IOException e) {
            System.err.println("[DEBUG] Failed to write to log file: " + e.getMessage());
        }
    }

        }

package ccsuite.webproxy;

import module java.base;
import ccsuite.webproxy.health.ErrorClassifier;
import ccsuite.webproxy.health.FailureType;

/**
 * What to do when a web turn fails before any output reached the client. The rule that matters: a turn that may have STARTED generating on the server
 * (a timeout, a connection cut while receiving) is charged to the account whether or not the client saw the end, so repeating it (on the same chat, in a
 * fresh chat, or on another account) pays for the same work again. Only a refusal that comes before any generation is worth another try.
 * (Found 2026-11-06: one timed-out Sonnet turn that wrote files was retried three times and took a free account from 15% to 100%.)
 */
public enum FollowUpFailure {
    /** A plain refusal ("Rate limited" 429, no exceeded_limit): nothing was generated; try the same chat once more after a short pause. */
    RETRY_SAME_CHAT,
    /** The account is used up: no retry and no fresh chat on it; fail over at once. */
    FAIL_OVER,
    /** A timeout or a cut: the server may be generating. Do not repeat it anywhere; tell the client. */
    DO_NOT_REPEAT,
    /** Anything else before output (refused, reset, 5xx): give the chat up and start a fresh one with the full conversation. */
    FRESH_CHAT,
    /** The request never left (could not connect, resolve or handshake): nothing was generated or charged, the chat is intact. Retry once on the same chat; the session is kept. */
    NOT_SENT;

    /** True when the failure came before the request was sent (connect, resolve, TLS handshake): nothing can be running on the server. */
    public static boolean notSent(Throwable t) {
        for (var e = t; e != null; e = e.getCause()) {
            var m = e.getMessage();
            if (m == null) continue;
            if (m.contains("Failed to connect") || m.contains("Could not connect") || m.contains("Could not resolve") || m.contains("SSL connect error")
                || m.contains("Connection refused") || m.contains("Connection timed out after")) return true;
        }
        return false;
    }

    /** A timeout or a cut AFTER the request was sent: the server may be generating and charging. */
    public static boolean mayHaveStarted(Throwable t) {
        return (timedOut(t) || cutWhileReceiving(t)) && !notSent(t);
    }

    /** The connection broke while the answer was coming in: the request had been accepted. */
    public static boolean cutWhileReceiving(Throwable t) {
        for (var e = t; e != null; e = e.getCause()) {
            var m = e.getMessage();
            if (m != null && (m.contains("Failure when receiving data") || m.contains("Connection closed abruptly") || m.contains("Recv failure"))) return true;
        }
        return false;
    }

    public static boolean timedOut(Throwable t) {
        for (var e = t; e != null; e = e.getCause()) {
            if (e instanceof java.net.SocketTimeoutException) return true;
            var m = e.getMessage();
            if (m != null && (m.contains("Timeout was reached") || m.toLowerCase().contains("timed out") || m.toLowerCase().contains("timeout"))) return true;
        }
        return false;
    }

    public static FollowUpFailure of(IOException failure) {
        if (notSent(failure)) return NOT_SENT;
        if (mayHaveStarted(failure)) return DO_NOT_REPEAT;
        return switch (ErrorClassifier.classify(failure)) {
            case QUOTA_EXHAUSTED -> FAIL_OVER;
            case RATE_LIMITED -> RETRY_SAME_CHAT;
            default -> FRESH_CHAT;
        };
    }
}

/**
 * Converts Anthropic Messages API format to CCAPI prompt format
 *
 * <p>Implements the collision-free XML tag strategy from PRP 02:
 * <ul>
 *   <li>Single-turn: Send prompt directly (with optional system prompt)</li>
 *   <li>Multi-turn: Use XML tags with numbered suffixes to avoid collisions</li>
 *   <li>Tags: custom_system_prompt, formatting_instructions, user, ai_assistant</li>
 *   <li>Algorithm: Scan all content, increment suffix if collision detected</li>
 * </ul>
 *
 * <p>Example multi-turn output:
 * <pre>
 * &lt;custom_system_prompt&gt;You are a Python expert.&lt;/custom_system_prompt&gt;
 * &lt;formatting_instructions&gt;Use XML tags. Respond as ai_assistant_2 without tags.&lt;/formatting_instructions&gt;
 * &lt;user&gt;What is 2+2?&lt;/user&gt;
 * &lt;ai_assistant&gt;4&lt;/ai_assistant&gt;
 * &lt;user_1&gt;What about 3+3?&lt;/user_1&gt;
 * </pre>
 *
 * <p>Example single-turn output:
 * <pre>
 * &lt;custom_system_prompt&gt;You are a Python expert.&lt;/custom_system_prompt&gt;
 *
 * What is 2+2?
 * </pre>
 */
public class MessageToPromptConverter {

    private static final String TAG_SYSTEM = "custom_system_prompt";
    private static final String TAG_FORMATTING = "formatting_instructions";
    private static final String TAG_USER = "user";
    private static final String TAG_ASSISTANT = "ai_assistant";
    /** Claude Code sends the environment (working directory, platform) and hook output as {@code system}-role messages inside {@code messages}. */
    private static final String TAG_SYSTEM_MESSAGE = "system_message";

    /**
     * Convert Anthropic request to CCAPI prompt format
     *
     * @param request the Anthropic API request
     * @return formatted prompt string for CCAPI
     */
    public static String convert(AnthropicRequest request) {
        var messages = request.getMessages();
        if (messages == null || messages.isEmpty()) {
            return "";
        }

        // Check if this is a single-turn request (optimization)
        if (isSingleTurn(messages)) {
            return convertSingleTurn(request);
        }

        // Multi-turn: Use XML formatting
        return convertMultiTurn(request);
    }

    /**
     * Check if this is a single-turn conversation
     * Single-turn = only 1 user message (no assistant responses)
     */
    private static boolean isSingleTurn(List<Message> messages) {
        // Count messages by role
        long userCount = messages.stream().filter(m -> "user".equals(m.getRole())).count();
        long assistantCount = messages.stream().filter(m -> "assistant".equals(m.getRole())).count();

        return userCount == 1 && assistantCount == 0;
    }

    /**
     * Convert single-turn request (optimized format)
     */
    private static String convertSingleTurn(AnthropicRequest request) {
        var result = new StringBuilder();
        var allContent = collectAllContent(request);

        // Add system prompt if present (with tools appended)
        var systemPrompt = buildSystemPromptWithTools(request);
        if (systemPrompt != null && !systemPrompt.isEmpty()) {
            var systemTag = findNonCollidingTag(TAG_SYSTEM, allContent);
            result.append("<").append(systemTag).append(">")
                  .append(systemPrompt)
                  .append("</").append(systemTag).append(">\n\n");
        }

        // The system-role messages (environment, hook output) are context for the one user message
        var systemMessageTag = findNonCollidingTag(TAG_SYSTEM_MESSAGE, allContent);
        for (var m : request.getMessages()) {
            if (!"system".equals(m.getRole())) continue;
            var text = contentOf(m);
            if (!text.isBlank()) result.append("<").append(systemMessageTag).append(">").append(text).append("</").append(systemMessageTag).append(">\n\n");
        }

        // Add the user message directly (no XML wrapper)
        var userMessage = request.getMessages().stream()
                .filter(m -> "user".equals(m.getRole()))
                .findFirst()
                .orElse(null);

        if (userMessage != null) {
            result.append(contentOf(userMessage));

            // Add prompt injection nudge if tools are available (PRP-15: prevent colon stop-sequence)
            if (request.getTools() != null && !request.getTools().isEmpty()) {
                result.append("\n\n")
                      .append(nudge());
            }
        }

        return result.toString();
    }

    /**
     * Convert multi-turn request (full XML formatting)
     */
    private static String convertMultiTurn(AnthropicRequest request) {
        var result = new StringBuilder();
        var messages = request.getMessages();
        var allContent = collectAllContent(request);

        // Determine the final assistant tag (for formatting instructions)
        int nextAssistantIndex = countMessagesWithRole(messages, "assistant");
        var finalAssistantTag = findNonCollidingTag(TAG_ASSISTANT, allContent, nextAssistantIndex);

        // Add system prompt if present (with tools appended)
        var systemPrompt = buildSystemPromptWithTools(request);
        if (systemPrompt != null && !systemPrompt.isEmpty()) {
            var systemTag = findNonCollidingTag(TAG_SYSTEM, allContent);
            result.append("<").append(systemTag).append(">")
                  .append(systemPrompt)
                  .append("</").append(systemTag).append(">\n\n");
        }

        // Add formatting instructions
        var formattingTag = findNonCollidingTag(TAG_FORMATTING, allContent);
        result.append("<").append(formattingTag).append(">")
              .append("This conversation uses XML-style tags for message boundaries.\n")
              .append("You are fulfilling the role of ").append(finalAssistantTag).append(".\n")
              .append(formattingRule())
              .append("</").append(formattingTag).append(">\n\n");

        // Add conversation messages with collision-free tags
        int userIndex = 0;
        int assistantIndex = 0;
        boolean isLastMessage = false;
        boolean hasTools = request.getTools() != null && !request.getTools().isEmpty();

        for (int i = 0; i < messages.size(); i++) {
            var message = messages.get(i);
            var content = contentOf(message);
            if (content.isEmpty()) {
                continue;
            }

            isLastMessage = (i == messages.size() - 1);

            if ("user".equals(message.getRole())) {
                var tag = findNonCollidingTag(TAG_USER, allContent, userIndex);
                result.append("<").append(tag).append(">")
                      .append(content);

                // Add prompt injection nudge to last user message if tools available (PRP-15)
                if (isLastMessage && hasTools) {
                    result.append("\n\n")
                          .append(nudge());
                }

                result.append("</").append(tag).append(">\n\n");
                userIndex++;
            } else if ("system".equals(message.getRole())) {
                var tag = findNonCollidingTag(TAG_SYSTEM_MESSAGE, allContent);
                result.append("<").append(tag).append(">").append(content);
                if (isLastMessage && hasTools) result.append("\n\n").append(nudge());
                result.append("</").append(tag).append(">\n\n");
            } else if ("assistant".equals(message.getRole())) {
                var tag = findNonCollidingTag(TAG_ASSISTANT, allContent, assistantIndex);
                result.append("<").append(tag).append(">")
                      .append(content)
                      .append("</").append(tag).append(">\n\n");
                assistantIndex++;
            }
        }

        return result.toString().trim();
    }

    /**
     * The prompt of a follow-up in a chat that already holds the earlier turns (PRP 74): only the new user message (tool results rendered as
     * {@code <tool_result>} blocks, new text as is) and the end-of-prompt reminder, which stays at the end of every message to keep the
     * format adherence. The tool instructions and the earlier assistant turns are already in the chat, so none of it is repeated.
     * Blank when the last message is not a user message with content.
     */
    public static String convertContinuation(AnthropicRequest request) {
        var messages = request.getMessages();
        return messages == null || messages.isEmpty() ? "" : convertContinuation(request, messages.size() - 1);
    }

    /**
     * Like {@link #convertContinuation(AnthropicRequest)} for the messages from {@code firstNew} to the end, all of which must be user messages: Claude Code follows
     * a prompt with attachments (hook output, reminders) as further user messages, and all of them are new. Blank when any of them is not a user message or
     * there is nothing in them.
     */
    public static String convertContinuation(AnthropicRequest request, int firstNew) {
        var messages = request.getMessages();
        if (messages == null || firstNew < 0 || firstNew >= messages.size()) return "";
        var parts = new java.util.ArrayList<String>();
        for (var i = firstNew; i < messages.size(); i++) {
            var role = messages.get(i).getRole();
            if (!"user".equals(role) && !"system".equals(role)) return "";
            var content = contentOf(messages.get(i));
            if (!content.isBlank()) parts.add("system".equals(role) ? "<" + TAG_SYSTEM_MESSAGE + ">" + content + "</" + TAG_SYSTEM_MESSAGE + ">" : content);
        }
        if (parts.isEmpty()) return "";
        return String.join("\n", parts) + "\n\n" + nudge();
    }

    private static String nudge() {
        return JsonToolFormat.enabled()
            ? JsonToolFormat.reminder()
            : "(If you need to use tools, please use the tool_use format provided above)";
    }

    private static String contentOf(Message message) {
        return JsonToolFormat.enabled() ? JsonToolFormat.render(message) : message.getTextContent();
    }

    private static String formattingRule() {
        return JsonToolFormat.enabled()
            ? "Respond as the assistant: plain text, plus tool_use blocks when you need a tool (see the tools section). Do not repeat the message tags."
            : "Respond with ONLY your answer as plain text.\nDo NOT include XML tags in your response.";
    }

    /**
     * Count messages with a specific role
     */
    private static int countMessagesWithRole(List<Message> messages, String role) {
        return (int) messages.stream()
                .filter(m -> role.equals(m.getRole()))
                .count();
    }

    /**
     * Collect all content from the request for collision detection
     */
    private static Set<String> collectAllContent(AnthropicRequest request) {
        var content = new HashSet<String>();

        // Add system prompt
        var systemPrompt = request.getSystemPrompt();
        if (systemPrompt != null) {
            content.add(systemPrompt);
        }

        // Add all message content
        if (request.getMessages() != null) {
            for (var message : request.getMessages()) {
                var text = message.getTextContent();
                if (!text.isEmpty()) {
                    content.add(text);
                }
            }
        }

        return content;
    }

    /**
     * Find a non-colliding tag name for the base tag
     * Checks if baseTag or &lt;baseTag&gt; appears in any content
     * If collision detected, tries baseTag_1, baseTag_2, etc.
     */
    private static String findNonCollidingTag(String baseTag, Set<String> allContent) {
        return findNonCollidingTag(baseTag, allContent, 0);
    }

    /**
     * Find a non-colliding tag name starting from a specific index
     * Useful for numbered tags in conversation history
     */
    private static String findNonCollidingTag(String baseTag, Set<String> allContent, int startIndex) {
        // For index 0, try the base tag without suffix first
        if (startIndex == 0) {
            if (!hasCollision(baseTag, allContent)) {
                return baseTag;
            }
        }

        // Try with numbered suffixes
        int index = Math.max(1, startIndex);
        while (index < 1000) { // Safety limit
            var candidateTag = baseTag + "_" + index;
            if (!hasCollision(candidateTag, allContent)) {
                return candidateTag;
            }
            index++;
        }

        // Fallback (should never happen with reasonable content)
        return baseTag + "_" + System.currentTimeMillis();
    }

    /**
     * Check if a tag collides with any content
     * A collision occurs if the tag name or &lt;tag&gt; or &lt;/tag&gt; appears in content
     */
    private static boolean hasCollision(String tag, Set<String> allContent) {
        var openTag = "<" + tag + ">";
        var closeTag = "</" + tag + ">";

        for (var content : allContent) {
            if (content.contains(tag) || content.contains(openTag) || content.contains(closeTag)) {
                return true;
            }
        }

        return false;
    }

    /**
     * Build system prompt with tools appended
     * Combines the original system prompt with tool definitions
     */
    private static String buildSystemPromptWithTools(AnthropicRequest request) {
        var systemPrompt = request.getSystemPrompt();
        var toolsText = ToolsConverter.convertToolsToText(request.getTools());

        if (systemPrompt == null || systemPrompt.isEmpty()) {
            return toolsText != null ? toolsText : "";
        }

        if (toolsText == null || toolsText.isEmpty()) {
            return systemPrompt;
        }

        return systemPrompt + "\n\n" + toolsText;
    }

    /**
     * Escape special characters in prompt text
     * Currently a no-op as we preserve content exactly, but provided for future use
     */
    private static String escapePromptText(String text) {
        // No escaping needed - content is preserved exactly as per design
        return text;
    }
}
package ccsuite.webproxy;

import ccsuite.webclient.*;

import jakarta.xml.bind.JAXBContext;
import jakarta.xml.bind.JAXBException;

import ccsuite.webproxy.config.Settings;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

/**
 * Manages loading and saving of CCAPIProxySettings.xml
 *
 * <p>Handles proxy-specific settings including:
 * <ul>
 *   <li>Debug logging configuration</li>
 *   <li>Conversation visibility settings</li>
 *   <li>Output limits and file logging</li>
 * </ul>
 */
public class ProxySettingsManager {

    private final Path settingsPath;

    /**
     * Create settings manager with default location
     * Default: <userhome>/xyz-jphil/ccapis/CCAPIProxySettings.xml
     */
    public ProxySettingsManager() {
        var userHome = System.getProperty("user.home");
        var settingsDir = Paths.get(userHome, "xyz-jphil", "ccapis");
        this.settingsPath = settingsDir.resolve("CCAPIProxySettings.xml");

        // Create directory if it doesn't exist
        try {
            Files.createDirectories(settingsDir);
        } catch (IOException e) {
            throw new RuntimeException("Failed to create settings directory: " + settingsDir, e);
        }
    }

    /**
     * Create settings manager with custom path
     */
    public ProxySettingsManager(Path settingsPath) {
        this.settingsPath = settingsPath;
        try {
            if (settingsPath.getParent() != null) {
                Files.createDirectories(settingsPath.getParent());
            }
        } catch (IOException e) {
            throw new RuntimeException("Failed to create settings directory: " + settingsPath.getParent(), e);
        }
    }

    /**
     * Create settings manager with custom path (string)
     */
    public ProxySettingsManager(String settingsPath) {
        this(Paths.get(settingsPath));
    }

    /**
     * Load proxy settings from XML file
     * Returns null if file doesn't exist (graceful degradation)
     */
    public Settings.ProxySettings load() {
        if (!Files.exists(settingsPath)) {
            System.out.println("[PROXY] Settings file not found: " + settingsPath);
            System.out.println("[PROXY] Debug logging will be disabled. Create CCAPIProxySettings.xml to enable.");
            return null;
        }

        try {
            var jaxbContext = JAXBContext.newInstance(Settings.ProxySettings.class);
            var unmarshaller = jaxbContext.createUnmarshaller();
            return (Settings.ProxySettings) unmarshaller.unmarshal(settingsPath.toFile());
        } catch (JAXBException e) {
            System.err.println("[PROXY] Failed to load settings from: " + settingsPath);
            System.err.println("[PROXY] Error: " + e.getMessage());
            e.printStackTrace();
            System.err.println("[PROXY] Debug logging will be disabled.");
            return null;
        }
    }

    /**
     * Save proxy settings to XML file
     */
    public void save(Settings.ProxySettings settings) throws JAXBException {
        var jaxbContext = JAXBContext.newInstance(Settings.ProxySettings.class);
        var marshaller = jaxbContext.createMarshaller();
        marshaller.setProperty(Marshaller.JAXB_FORMATTED_OUTPUT, true);
        marshaller.marshal(settings, settingsPath.toFile());
    }

    /**
     * Get the settings file path
     */
    public Path getSettingsPath() {
        return settingsPath;
    }

    /**
     * Check if settings file exists
     */
    public boolean exists() {
        return Files.exists(settingsPath);
    }
}
package ccsuite.webproxy;

import ccsuite.webclient.*;

import lombok.Data;
import lombok.experimental.Accessors;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.time.Instant;
import java.time.ZoneId;
import java.time.format.DateTimeFormatter;
import java.util.Map;
import java.util.Properties;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * Logs individual HTTP requests and responses as separate files.
 *
 * <p>For each request/response pair, creates:
 * <ul>
 *   <li>NNNN-req.header - Raw request headers</li>
 *   <li>NNNN-req.body - Raw request body</li>
 *   <li>NNNN-res.header - Raw response headers</li>
 *   <li>NNNN-res.body - Raw response body</li>
 *   <li>NNNN-meta.properties - Metadata (URL, timing, status, etc.)</li>
 * </ul>
 *
 * <p>Also maintains an index.txt file with one-line summaries for quick scanning.
 *
 * <p>File timestamps (creation/modification) provide timing information.
 */
public class RequestResponseFileLogger {

    private final Path logsDirectory;
    private final AtomicInteger sequenceNumber;
    private final boolean createIndexFile;

    private static final DateTimeFormatter TIMESTAMP_FORMAT =
        DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss.SSS")
                         .withZone(ZoneId.of("UTC"));

    /**
     * Create a new request/response file logger
     *
     * @param logsDirectory directory where files will be saved
     * @param createIndexFile whether to maintain an index.txt summary file
     */
    public RequestResponseFileLogger(Path logsDirectory, boolean createIndexFile) {
        this.logsDirectory = logsDirectory;
        this.sequenceNumber = new AtomicInteger(0);
        this.createIndexFile = createIndexFile;

        try {
            Files.createDirectories(logsDirectory);
        } catch (IOException e) {
            System.err.println("[RequestResponseFileLogger] Failed to create logs directory: " + e.getMessage());
        }

        if (createIndexFile) {
            initializeIndexFile();
        }
    }

    /**
     * Initialize the index file with header
     */
    private void initializeIndexFile() {
        var indexFile = logsDirectory.resolve("index.txt");
        try {
            var header = "# Request/Response Index - Created " + TIMESTAMP_FORMAT.format(Instant.now()) + " UTC\n" +
                        "# Format: [NNNN] Timestamp | Method | URL | Status | Size\n" +
                        "# " + "=".repeat(100) + "\n";
            Files.writeString(indexFile, header, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
        } catch (IOException e) {
            System.err.println("[RequestResponseFileLogger] Failed to initialize index file: " + e.getMessage());
        }
    }

    /**
     * Log a complete request/response transaction
     *
     * @param request the request information
     */
    public void logTransaction(Transaction request) {
        var seq = sequenceNumber.getAndIncrement();
        var seqStr = String.format("%04d", seq);

        try {
            // Write request files
            writeFile(seqStr + "-req.header", request.requestHeaders);
            writeFile(seqStr + "-req.body", request.requestBody != null ? request.requestBody : "");

            // Write response files
            writeFile(seqStr + "-res.header", request.responseHeaders != null ? request.responseHeaders : "");
            writeFile(seqStr + "-res.body", request.responseBody != null ? request.responseBody : "");

            // Write metadata
            writeMetadata(seqStr, request);

            // Update index
            if (createIndexFile) {
                updateIndex(seqStr, request);
            }

        } catch (IOException e) {
            System.err.println("[RequestResponseFileLogger] Failed to log transaction " + seqStr + ": " + e.getMessage());
        }
    }

    /**
     * Write a file with the given name and content
     */
    private void writeFile(String fileName, String content) throws IOException {
        var filePath = logsDirectory.resolve(fileName);
        Files.writeString(filePath, content, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
    }

    /**
     * Write metadata properties file
     */
    private void writeMetadata(String seqStr, Transaction tx) throws IOException {
        var props = new Properties();

        // Request metadata
        props.setProperty("request.method", tx.method != null ? tx.method : "UNKNOWN");
        props.setProperty("request.url", tx.url != null ? tx.url : "");
        props.setProperty("request.timestamp", tx.requestTimestamp != null ? tx.requestTimestamp : "");

        // Response metadata
        props.setProperty("response.status", String.valueOf(tx.responseStatus));
        props.setProperty("response.timestamp", tx.responseTimestamp != null ? tx.responseTimestamp : "");

        // Timing
        if (tx.durationMs != null) {
            props.setProperty("duration.ms", String.valueOf(tx.durationMs));
        }

        // Sizes
        props.setProperty("request.body.size", String.valueOf(tx.requestBody != null ? tx.requestBody.length() : 0));
        props.setProperty("response.body.size", String.valueOf(tx.responseBody != null ? tx.responseBody.length() : 0));

        // Domain/Host extraction
        if (tx.url != null) {
            try {
                var uri = java.net.URI.create(tx.url);
                props.setProperty("request.host", uri.getHost() != null ? uri.getHost() : "");
                props.setProperty("request.path", uri.getPath() != null ? uri.getPath() : "");
            } catch (Exception e) {
                // Ignore URI parsing errors
            }
        }

        // Additional metadata
        if (tx.metadata != null) {
            tx.metadata.forEach(props::setProperty);
        }

        // Write properties file
        var metaPath = logsDirectory.resolve(seqStr + "-meta.properties");
        try (var out = Files.newOutputStream(metaPath, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING)) {
            props.store(out, "Request/Response Metadata - Transaction " + seqStr);
        }
    }

    /**
     * Update the index file with a summary line
     */
    private void updateIndex(String seqStr, Transaction tx) {
        var indexFile = logsDirectory.resolve("index.txt");
        try {
            var method = tx.method != null ? tx.method : "???";
            var url = tx.url != null ? tx.url : "???";
            var status = tx.responseStatus != 0 ? String.valueOf(tx.responseStatus) : "???";
            var size = tx.responseBody != null ? tx.responseBody.length() : 0;
            var timestamp = tx.requestTimestamp != null ? tx.requestTimestamp : "???";

            // Truncate URL if too long
            if (url.length() > 80) {
                url = url.substring(0, 77) + "...";
            }

            var summary = String.format("[%s] %s | %-6s | %-80s | %s | %d bytes\n",
                                       seqStr, timestamp, method, url, status, size);

            Files.writeString(indexFile, summary, StandardOpenOption.CREATE, StandardOpenOption.APPEND);
        } catch (IOException e) {
            System.err.println("[RequestResponseFileLogger] Failed to update index: " + e.getMessage());
        }
    }

    /**
     * Log the converted Claude-Code-compatible response (what was sent back to Claude Code)
     *
     * @param seqStr sequence number string (e.g., "0004")
     * @param anthropicResponseJson JSON string of the converted Anthropic response
     */
    public void logConvertedAnthropicResponse(String seqStr, String anthropicResponseJson) {
        try {
            writeFile(seqStr + "-ccr.body", anthropicResponseJson);
        } catch (IOException e) {
            System.err.println("[RequestResponseFileLogger] Failed to log converted response: " + e.getMessage());
        }
    }

    /**
     * Get the current sequence number (useful for testing/debugging)
     */
    public int getCurrentSequence() {
        return sequenceNumber.get();
    }

    /**
     * Get the logs directory (conversation directory)
     */
    public Path getLogsDirectory() {
        return logsDirectory;
    }

    /**
     * Represents a complete HTTP transaction (request + response)
     */
    @Data
    @Accessors(fluent = true, chain = true)
    public static class Transaction {
        String method;
        String url;
        String requestHeaders;
        String requestBody;
        String requestTimestamp;

        String responseHeaders;
        String responseBody;
        int responseStatus;
        String responseTimestamp;

        Long durationMs;
        Map<String, String> metadata;
    }
}
package ccsuite.webproxy;

import ccsuite.webclient.*;

import okhttp3.Interceptor;
import okhttp3.Request;
import okhttp3.Response;
import okhttp3.ResponseBody;
import okio.Buffer;

import java.io.IOException;
import java.time.Instant;
import java.time.ZoneId;
import java.time.format.DateTimeFormatter;
import java.util.HashMap;

/**
 * OkHttp interceptor that logs requests and responses to individual files
 * via RequestResponseFileLogger.
 *
 * <p>Captures:
 * <ul>
 *   <li>Request headers and body</li>
 *   <li>Response headers and body</li>
 *   <li>Timing information</li>
 *   <li>Status codes</li>
 * </ul>
 */
public class RequestResponseLoggingInterceptor implements Interceptor {

    private final RequestResponseFileLogger fileLogger;

    private static final DateTimeFormatter TIMESTAMP_FORMAT =
        DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss.SSS")
                         .withZone(ZoneId.of("UTC"));

    public RequestResponseLoggingInterceptor(RequestResponseFileLogger fileLogger) {
        this.fileLogger = fileLogger;
    }

    @Override
    public Response intercept(Chain chain) throws IOException {
        var request = chain.request();
        var startTime = System.currentTimeMillis();
        var requestTimestamp = TIMESTAMP_FORMAT.format(Instant.now());

        // Capture request details
        var requestHeaders = captureRequestHeaders(request);
        var requestBody = captureRequestBody(request);

        // Execute request
        Response response;
        String responseBody = null;
        String responseHeaders = null;
        int responseStatus = 0;

        try {
            response = chain.proceed(request);
            responseStatus = response.code();

            // Capture response details
            responseHeaders = captureResponseHeaders(response);

            // Read and cache response body
            var originalBody = response.body();
            if (originalBody != null) {
                responseBody = originalBody.string();

                // Recreate response with buffered body
                response = response.newBuilder()
                    .body(ResponseBody.create(responseBody, originalBody.contentType()))
                    .build();
            }

        } catch (IOException e) {
            // Log failed request
            logTransaction(request, requestTimestamp, requestHeaders, requestBody,
                          null, null, 0, startTime);
            throw e;
        }

        // Log successful transaction
        logTransaction(request, requestTimestamp, requestHeaders, requestBody,
                      responseHeaders, responseBody, responseStatus, startTime);

        return response;
    }

    /**
     * Log the complete transaction
     */
    private void logTransaction(Request request, String requestTimestamp,
                               String requestHeaders, String requestBody,
                               String responseHeaders, String responseBody,
                               int responseStatus, long startTime) {
        var endTime = System.currentTimeMillis();
        var responseTimestamp = TIMESTAMP_FORMAT.format(Instant.now());
        var duration = endTime - startTime;

        var metadata = new HashMap<String, String>();
        metadata.put("request.scheme", request.url().scheme());

        var transaction = new RequestResponseFileLogger.Transaction()
            .method(request.method())
            .url(request.url().toString())
            .requestHeaders(requestHeaders)
            .requestBody(requestBody)
            .requestTimestamp(requestTimestamp)
            .responseHeaders(responseHeaders)
            .responseBody(responseBody)
            .responseStatus(responseStatus)
            .responseTimestamp(responseTimestamp)
            .durationMs(duration)
            .metadata(metadata);

        fileLogger.logTransaction(transaction);
    }

    /**
     * Capture request headers as string
     */
    private String captureRequestHeaders(Request request) {
        var sb = new StringBuilder();
        sb.append(request.method()).append(" ").append(request.url()).append("\n");

        var headers = request.headers();
        for (var i = 0; i < headers.size(); i++) {
            sb.append(headers.name(i)).append(": ").append(headers.value(i)).append("\n");
        }

        return sb.toString();
    }

    /**
     * Capture request body as string
     */
    private String captureRequestBody(Request request) {
        try {
            var requestCopy = request.newBuilder().build();
            var buffer = new Buffer();
            var body = requestCopy.body();

            if (body != null) {
                body.writeTo(buffer);
                return buffer.readUtf8();
            }
        } catch (IOException e) {
            return "[Error reading request body: " + e.getMessage() + "]";
        }

        return "";
    }

    /**
     * Capture response headers as string
     */
    private String captureResponseHeaders(Response response) {
        var sb = new StringBuilder();
        // Use actual protocol from response (could be HTTP/1.1, HTTP/2, etc.)
        sb.append(response.protocol()).append(" ").append(response.code()).append(" ").append(response.message()).append("\n");

        var headers = response.headers();
        for (var i = 0; i < headers.size(); i++) {
            sb.append(headers.name(i)).append(": ").append(headers.value(i)).append("\n");
        }

        return sb.toString();
    }
}

import ccsuite.webproxy.anthropic.AnthropicRequest;

import java.io.IOException;

/**
 * The input tokens of a request, estimated from its content (PRP 85): the features of {@link TokenFeatures} times weights fitted against the paid count_tokens API on
 * real Claude Code conversations ({@code token-weights.json}, written by {@code ccacc tokencheck model fit}, refreshed by {@code ccacc tokencheck model review}).
 * Measured on whole conversations the characters-over-one-ratio estimate was 13% off on average and the content estimate 4%, with the total within about 1%.
 *
 * <p>No API is called here (PRP 83, R3): it is a handful of regular-expression passes over the request. The estimate is scaled by the file's safety factor so it errs
 * high, because under-reporting how full the context is is the failure that matters. Anything that goes wrong (a request that cannot be read, no weights file) falls
 * back to the one ratio of {@link TokenEstimate}.
 */
public final class RequestTokens {
    private static final ObjectMapper JSON = new ObjectMapper();
    static final Weights WEIGHTS = Weights.load();

    private RequestTokens() { }

    /** The fitted weights by feature name, the framing constants and the safety factor. */
    record Weights(Map<String, Double> weights, double base, double toolsFixed, double perToolDef, double safety) {
        static Weights load() {
            try (var in = RequestTokens.class.getResourceAsStream("/ccsuite/webproxy/token-weights.json")) {
                if (in == null) return null;
                var j = JSON.readTree(in);
                var w = new java.util.LinkedHashMap<String, Double>();
                j.path("weights").fields().forEachRemaining(e -> w.put(e.getKey(), e.getValue().asDouble()));
                var fr = j.path("framing");
                return new Weights(w, fr.path("base").asDouble(), fr.path("toolsFixed").asDouble(), fr.path("perToolDef").asDouble(), j.path("safety").asDouble(1.0));
            } catch (IOException | RuntimeException e) {
                return null;
            }
        }
    }

    /** The estimate for a request given as its parts; the fallback characters are used only if the estimate cannot be made. */
    public static int of(JsonNode messages, JsonNode system, List<JsonNode> tools, long fallbackChars) {
        try {
            if (WEIGHTS == null || messages == null || !messages.isArray() || messages.isEmpty()) return TokenEstimate.tokens(fallbackChars);   // nothing to read: the one ratio
            var f = TokenFeatures.ofRequest(messages, system, tools);
            var t = WEIGHTS.base();
            for (var e : WEIGHTS.weights().entrySet()) t += e.getValue() * f.getOrDefault(e.getKey(), 0.0);
            var defs = f.getOrDefault("tool.defs", 0.0);
            if (defs > 0) t += WEIGHTS.toolsFixed() + WEIGHTS.perToolDef() * defs;
            return t <= 0 ? TokenEstimate.tokens(fallbackChars) : (int) Math.ceil(t * WEIGHTS.safety());
        } catch (RuntimeException e) {
            return TokenEstimate.tokens(fallbackChars);
        }
    }

    /** The estimate for a parsed request. */
    public static int of(AnthropicRequest request, long fallbackChars) {
        try {
            return of(JSON.valueToTree(request.getMessages()), request.getSystem(), request.getTools(), fallbackChars);
        } catch (RuntimeException e) {
            return TokenEstimate.tokens(fallbackChars);
        }
    }

    /** The estimate for a request body as received (the guard has the bytes, not the parsed request); the fallback is the body's own length. */
    public static int ofBody(byte[] body) {
        try {
            var j = JSON.readTree(body);
            var tools = new java.util.ArrayList<JsonNode>();
            j.path("tools").forEach(tools::add);
            return of(j.path("messages"), j.path("system"), tools, body.length);
        } catch (IOException | RuntimeException e) {
            return TokenEstimate.tokens(body.length);
        }
    }
}

/**
 * Manages system tray icon for CCAPI Proxy Server.
 *
 * <p>Provides:
 * <ul>
 *   <li>System tray icon with status indicator</li>
 *   <li>Right-click menu with server information and exit option</li>
 *   <li>Graceful shutdown when exit is clicked</li>
 *   <li>Headless-safe operation (won't fail if system tray not supported)</li>
 * </ul>
 *
 * <p>Usage:
 * <pre>
 * var trayManager = new SystemTrayManager(8080);
 * trayManager.show(() -> {
 *     // Cleanup code before exit
 *     proxy.stop();
 *     System.exit(0);
 * });
 * </pre>
 */
public class SystemTrayManager {

    private final int port;
    private TrayIcon trayIcon;
    private boolean isShown = false;

    /**
     * Create system tray manager for proxy on given port
     */
    public SystemTrayManager(int port) {
        this.port = port;
    }

    /**
     * Show system tray icon with exit handler
     *
     * @param onExit Runnable to execute when user clicks Exit (should call proxy.stop() and System.exit(0))
     * @return true if system tray is supported and icon was shown, false otherwise
     */
    public boolean show(Runnable onExit) {
        // Check if system tray is supported
        if (!SystemTray.isSupported()) {
            System.err.println("[SYSTEM TRAY] System tray is not supported on this platform");
            return false;
        }

        // Check if we're running in headless mode
        if (GraphicsEnvironment.isHeadless()) {
            System.err.println("[SYSTEM TRAY] Running in headless mode, system tray unavailable");
            return false;
        }

        try {
            // Get the system tray
            var systemTray = SystemTray.getSystemTray();

            // Create tray icon image (simple colored circle)
            var image = createTrayIconImage();

            // Create popup menu
            var popup = new PopupMenu();

            // Add menu items
            var titleItem = new MenuItem("CCAPI Proxy Server");
            titleItem.setEnabled(false); // Make it non-clickable (just info)
            popup.add(titleItem);

            popup.addSeparator();

            var statusItem = new MenuItem("Status: Running");
            statusItem.setEnabled(false);
            popup.add(statusItem);

            var portItem = new MenuItem("Port: " + port);
            portItem.setEnabled(false);
            popup.add(portItem);

            popup.addSeparator();

            // Add exit option
            var exitItem = new MenuItem("Exit");
            exitItem.addActionListener(e -> {
                System.out.println("\n[SYSTEM TRAY] Exit requested by user");
                if (onExit != null) {
                    onExit.run();
                }
            });
            popup.add(exitItem);

            // Create tray icon
            trayIcon = new TrayIcon(image, "CCAPI Proxy Server (Port: " + port + ")", popup);
            trayIcon.setImageAutoSize(true);

            // Add double-click handler (shows info)
            trayIcon.addActionListener(e -> {
                trayIcon.displayMessage(
                    "CCAPI Proxy Server",
                    "Running on port " + port + "\nRight-click for options",
                    TrayIcon.MessageType.INFO
                );
            });

            // Add to system tray
            systemTray.add(trayIcon);

            isShown = true;
            System.out.println("[SYSTEM TRAY] System tray icon added successfully");

            // Show initial notification
            trayIcon.displayMessage(
                "CCAPI Proxy Server",
                "Server started on port " + port,
                TrayIcon.MessageType.INFO
            );

            return true;

        } catch (AWTException e) {
            System.err.println("[SYSTEM TRAY] Failed to add system tray icon: " + e.getMessage());
            e.printStackTrace();
            return false;
        }
    }

    /**
     * Remove the system tray icon
     */
    public void hide() {
        if (isShown && trayIcon != null) {
            try {
                SystemTray.getSystemTray().remove(trayIcon);
                isShown = false;
                System.out.println("[SYSTEM TRAY] System tray icon removed");
            } catch (Exception e) {
                System.err.println("[SYSTEM TRAY] Error removing tray icon: " + e.getMessage());
            }
        }
    }

    /**
     * Display a notification message
     */
    public void showNotification(String title, String message, TrayIcon.MessageType type) {
        if (isShown && trayIcon != null) {
            trayIcon.displayMessage(title, message, type);
        }
    }

    /**
     * Create a simple colored circle icon for the system tray
     */
    private BufferedImage createTrayIconImage() {
        // Get tray icon size (usually 16x16 or 32x32 depending on OS)
        var trayIconSize = SystemTray.getSystemTray().getTrayIconSize();
        var size = Math.max(16, Math.min(trayIconSize.width, trayIconSize.height));

        var image = new BufferedImage(size, size, BufferedImage.TYPE_INT_ARGB);
        var g2d = image.createGraphics();

        // Enable antialiasing for smoother circle
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // Draw green circle (indicates server is running)
        g2d.setColor(new Color(76, 175, 80)); // Material Green
        g2d.fillOval(2, 2, size - 4, size - 4);

        // Draw white border
        g2d.setColor(Color.WHITE);
        g2d.setStroke(new BasicStroke(1.5f));
        g2d.drawOval(2, 2, size - 4, size - 4);

        g2d.dispose();
        return image;
    }

    /**
     * Check if system tray is supported on this platform
     */
    public static boolean isSupported() {
        return !GraphicsEnvironment.isHeadless() && SystemTray.isSupported();
    }

/**
 * The one place where characters become tokens (PRP 83).
 *
 * <p>The ratio is empirical, not computed: no counting API is called on the request path. It was measured on real sessions with the current
 * tokenizer: Java and prose about 2.6 characters per token, JSON about 2.0, and a 2.45-million-character test about 2.8.
 * 2.4 is the FALLBACK ratio, used only when {@link RequestTokens} cannot read a request, and for the reply text (PRP 85). Against the paid count_tokens API on whole conversations the characters-over-one-ratio estimate was 13% off on average whatever the ratio (the spread per conversation is 2.2 to 2.9 characters per token); at 2.4 it is +3% in total with a third of conversations slightly under-counted. An estimate that is too big is harmless, one that is too small
 * hides how full the context is, so the error is kept on the conservative side. The old tokenizer gave about 4, which made every figure
 * here too small.
 *
 * <p>Used for: the {@code input_tokens} and {@code output_tokens} of the usage block Claude Code reads, and the input figure of the
 * predicted-quota record. Nothing else in the proxy turns characters into tokens.
 */
public final class TokenEstimate {
    // DO NOT CHANGE THIS VALUE WITHOUT THINKING. It is deliberately LOW (conservative), and raising it makes the free lane lie to Claude Code.
    //
    // What it does: Claude Code reads the size of its session from the usage block this proxy writes, and decides when to compact from it.
    // Characters / this value = the tokens we report. A LOWER value reports MORE tokens, so a session looks fuller than it may be and
    // compaction comes early. A HIGHER value reports FEWER tokens, a session looks emptier than it is, and it runs past the real limit.
    // Too many tokens is harmless; too few is the failure this PRP (83) was written to remove, so when unsure, round DOWN.
    //
    // Where 2.4 comes from (measured on real sessions with the current tokenizer, characters per token): Java and prose about 2.6,
    // JSON about 2.0, a 2.45-million-character test about 2.8. The plain average is about 2.5. The user decided on 2026-11-07 to take a
    // value slightly BELOW that average. It was 2.3, then 2.25 (interim, 2026-11-07), and became 2.4 on 2026-11-08 as the fallback behind the content estimate (PRP 85: on 255 whole conversations counted by the paid API the characters per real token were 2.47 in total, 2.22 to 2.88 per conversation; 2.4 is +3% in total). The old tokenizer gave about 4, which is no longer true.
    //
    // To change it: re-measure on fresh sessions (JSON-heavy tool results pull the real ratio towards 2.0), keep the result at or
    // below the measured average, and update the expected numbers in TokenEstimateTest and ChatReuseFlowTest (they pin this value).
    // Never call a counting API on the request path to avoid this estimate; that would slow every turn (PRP 83, R3).
    public static final double CHARS_PER_TOKEN = 2.4;

    private TokenEstimate() { }

    /** ceil(characters / {@link #CHARS_PER_TOKEN}); 0 for no characters. */
    public static int tokens(long characters) {
        return characters <= 0 ? 0 : (int) Math.ceil(characters / CHARS_PER_TOKEN);
    }

    public static int tokens(CharSequence text) {
        return text == null ? 0 : tokens(text.length());
    }
}
package ccsuite.webproxy;

import com.fasterxml.jackson.databind.JsonNode;

import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.regex.Pattern;

/**
 * The numbers a token estimate is made from (PRP 85): for a request, the characters of every block by the class of the block (prose, code, json, numbered file
 * listing) and the same text counted by character kind (words, digits, punctuation, whitespace, capitals, non-ASCII). The class is found from the block's own text;
 * nothing here calls an API or a tokenizer. {@link RequestTokens} turns these into a token estimate with weights fitted against the paid count_tokens API.
 * The measurement and the fit live in ccacc ({@code ccacc tokencheck model}); this class is shared so the measurement and the request path compute the same features.
 */
public final class TokenFeatures {
    public static final List<String> CLASSES = List.of("prose", "code", "json", "numbered");
    static final Pattern WORD = Pattern.compile("\\p{L}+"), DIGITS = Pattern.compile("\\d+"), PUNCT = Pattern.compile("[^\\p{L}\\p{N}\\s]+"), NUMBERED = Pattern.compile("^\\s*\\d+[→\\t]");
    static final String SYMBOLS = "{}()[];=<>\"'\\/:,|&*#$@%^~`";
    public static final List<String> SCRIPTS = List.of("latinx", "cyrillic", "indic", "cjk", "symbols", "emoji", "otherscript");

    private TokenFeatures() { }

    /** What a block's text is: json (a value, or a tool call's input), numbered (a file shown with line numbers), code (dense in symbols) or prose. */
    public static String classOf(String text, boolean toolInput) {
        var t = text.strip();
        if (toolInput || (t.length() > 1 && (t.startsWith("{") && t.endsWith("}") || t.startsWith("[") && t.endsWith("]")))) return "json";
        var lines = text.lines().limit(40).toList();
        if (!lines.isEmpty() && lines.stream().filter(l -> NUMBERED.matcher(l).find()).count() * 2 >= lines.size()) return "numbered";
        var symbols = t.chars().filter(c -> SYMBOLS.indexOf(c) >= 0).count();
        return t.isEmpty() || symbols * 1.0 / t.length() <= 0.08 ? "prose" : "code";
    }

    /** The features of a request's messages, system prompt and tool definitions (the last two may be null). */
    public static Map<String, Double> ofRequest(JsonNode messages, JsonNode system, List<JsonNode> tools) {
        var f = of(messages);
        if (system != null && !system.isNull()) {
            if (system.isTextual()) text(f, system.asText(), false);
            else for (var b : system) text(f, b.path("text").asText(), false);
        }
        if (tools != null) for (var t : tools) { text(f, t.toString(), true); add(f, "tool.defs", 1); }
        return f;
    }

    /** All features of a Messages array, by name. */
    public static Map<String, Double> of(JsonNode messages) {
        var f = new LinkedHashMap<String, Double>();
        if (messages == null) return f;
        for (var m : messages) {
            add(f, "messages", 1);
            var content = m.path("content");
            if (content.isTextual()) { add(f, "blocks", 1); add(f, "b.text", 1); text(f, content.asText(), false); continue; }
            for (var b : content) {
                add(f, "blocks", 1);
                add(f, "b." + b.path("type").asText(), 1);
                switch (b.path("type").asText()) {
                    case "text" -> text(f, b.path("text").asText(), false);
                    case "thinking" -> text(f, b.path("thinking").asText(), false);
                    case "tool_use" -> { text(f, b.path("name").asText(), false); text(f, b.path("input").toString(), true); }
                    case "tool_result" -> {
                        var c = b.path("content");
                        if (c.isTextual()) text(f, c.asText(), false);
                        else for (var x : c) text(f, x.path("text").asText(), false);
                    }
                    default -> { }
                }
            }
        }
        return f;
    }

    static void text(Map<String, Double> f, String s, boolean toolInput) {
        var c = classOf(s, toolInput);
        var words = WORD.matcher(s).results().toList();
        var wordChars = words.stream().mapToInt(r -> r.end() - r.start()).sum();
        var digitRuns = DIGITS.matcher(s).results().toList();
        var digits = digitRuns.stream().mapToInt(r -> r.end() - r.start()).sum();
        var puncts = PUNCT.matcher(s).results().toList();
        var punctChars = puncts.stream().mapToInt(r -> r.end() - r.start()).sum();
        var ws = (int) s.chars().filter(Character::isWhitespace).count();
        var nl = (int) s.chars().filter(x -> x == '\n').count();
        var upper = (int) s.chars().filter(Character::isUpperCase).count();
        var nonAscii = (int) s.chars().filter(x -> x > 127).count();
        s.codePoints().filter(x -> x > 127).forEach(x -> add(f, script(x), 1));
        add(f, "chars", s.length()); add(f, "words", words.size()); add(f, "wordchars", wordChars); add(f, "digitruns", digitRuns.size()); add(f, "digits", digits);
        add(f, "punctruns", puncts.size()); add(f, "punctchars", punctChars); add(f, "ws", ws); add(f, "nl", nl); add(f, "upper", upper); add(f, "nonascii", nonAscii);
        add(f, c + ".chars", s.length()); add(f, c + ".words", words.size()); add(f, c + ".wordchars", wordChars); add(f, c + ".digits", digits); add(f, c + ".punctruns", puncts.size()); add(f, c + ".ws", ws);
    }

    /** The group a non-ASCII character belongs to: tokens per character differ a lot between them. */
    public static String script(int c) {
        if (c < 0x250) return "latinx";
        if (c >= 0x370 && c < 0x530) return "cyrillic";
        if (c >= 0x900 && c < 0xE00) return "indic";
        if (c >= 0x2000 && c < 0x2C00) return "symbols";
        if (c >= 0x3000 && c < 0xA000 || c >= 0xAC00 && c < 0xD7B0) return "cjk";
        if (c >= 0x1F000) return "emoji";
        return "otherscript";
    }

    static void add(Map<String, Double> f, String key, double v) { f.merge(key, v, Double::sum); }

    public static double[] vector(Map<String, Double> f, List<String> names) { return names.stream().mapToDouble(n -> f.getOrDefault(n, 0.0)).toArray(); }
}

/**
 * Converts Anthropic API tool definitions to text format for CCAPI
 *
 * <p>Anthropic API uses structured JSON for tools, but CCAPI only accepts plain text.
 * This converter transforms tool definitions into XML-style text that Claude can understand.
 */
public class ToolsConverter {

    /**
     * Convert Anthropic tools format to XML text description
     *
     * @param tools List of tool definitions in Anthropic JSON format
     * @return Text description of tools in XML format, or empty string if no tools
     */
    public static String convertToolsToText(List<JsonNode> tools) {
        if (tools == null || tools.isEmpty()) {
            return "";
        }

        if (JsonToolFormat.enabled()) {
            return JsonToolFormat.toolsBlock(tools);
        }

        var result = new StringBuilder();
        result.append("# Available Tools\n\n");
        result.append("IMPORTANT: You MUST use <tool_use> tags (Anthropic format). Do NOT use <invoke> or <use_tool> tags.\n\n");
        result.append("Use tools by outputting XML in EXACTLY this format:\n");
        result.append(TOOL_CALL_EXAMPLE).append("\n\n");
        result.append("CRITICAL: Use <tool_use name=\"...\"> with the standard Anthropic format. The tag name MUST be 'tool_use'.\n\n");

        // Convert each tool
        for (var tool : tools) {
            convertSingleTool(tool, result);
        }

        return result.toString();
    }

    /**
     * Convert a single tool definition to text
     */
    private static void convertSingleTool(JsonNode tool, StringBuilder result) {
        // Get tool name
        var name = tool.has("name") ? tool.get("name").asText() : "unknown";
        result.append("## Tool: ").append(name).append("\n\n");

        // Get description
        if (tool.has("description")) {
            result.append("**Description:** ").append(tool.get("description").asText()).append("\n\n");
        }

        // Get input schema (parameters)
        if (tool.has("input_schema")) {
            convertInputSchema(tool.get("input_schema"), result);
        }

        result.append("\n");
    }

    /**
     * Convert input schema (parameters) to text
     */
    private static void convertInputSchema(JsonNode schema, StringBuilder result) {
        result.append("**Parameters:**\n\n");

        if (!schema.has("properties")) {
            result.append("  (No parameters)\n");
            return;
        }

        var properties = schema.get("properties");
        var required = schema.has("required") ? schema.get("required") : null;

        // Convert each parameter
        Iterator<Map.Entry<String, JsonNode>> fields = properties.fields();
        while (fields.hasNext()) {
            var field = fields.next();
            var paramName = field.getKey();
            var paramDef = field.getValue();

            result.append("  - `").append(paramName).append("`");

            // Check if required
            boolean isRequired = false;
            if (required != null && required.isArray()) {
                for (var req : required) {
                    if (req.asText().equals(paramName)) {
                        isRequired = true;
                        break;
                    }
                }
            }

            if (isRequired) {
                result.append(" **(required)**");
            }

            // Add type
            if (paramDef.has("type")) {
                result.append(" - Type: `").append(paramDef.get("type").asText()).append("`");
            }

            result.append("\n");

            // Add description
            if (paramDef.has("description")) {
                result.append("    ").append(paramDef.get("description").asText()).append("\n");
            }

            // Add enum values if present
            if (paramDef.has("enum")) {
                result.append("    Allowed values: ");
                var enumValues = paramDef.get("enum");
                for (int i = 0; i < enumValues.size(); i++) {
                    if (i > 0) result.append(", ");
                    result.append("`").append(enumValues.get(i).asText()).append("`");
                }
                result.append("\n");
            }

            result.append("\n");
        }
    }
}

    List<Message> messages;

    /**
     * System prompt (optional)
     * Can be either:
     * - A simple string: "You are a helpful assistant"
     * - An array of content blocks: [{"type":"text","text":"You are..."}]
     */
    JsonNode system;

    /**
     * Maximum tokens to generate
     */
    @JsonProperty("max_tokens")
    Integer maxTokens;

    /**
     * Temperature (NOT SUPPORTED by CCAPI - will be ignored)
     * Included for API compatibility
     */
    Double temperature;

    /**
     * Whether to stream the response
     */
    Boolean stream;

    /**
     * Additional metadata (optional)
     */
    Object metadata;

    /**
     * Stop sequences (optional)
     */
    @JsonProperty("stop_sequences")
    List<String> stopSequences;

    /**
     * Top-k sampling (optional)
     */
    @JsonProperty("top_k")
    Integer topK;

    /**
     * Top-p sampling (optional)
     */
    @JsonProperty("top_p")
    Double topP;

    /**
     * Tool definitions (optional)
     * Array of tool definitions that Claude can use
     */
    List<JsonNode> tools;

    /**
     * Tool choice configuration (optional)
     * Controls how Claude uses tools: "auto", "any", "none", or {"type":"tool","name":"tool_name"}
     */
    @JsonProperty("tool_choice")
    JsonNode toolChoice;

    /**
     * Extract system prompt as a string
     * Handles both string and array formats
     */
    public String getSystemPrompt() {
        if (system == null) {
            return null;
        }

        // If it's a simple string
        if (system.isTextual()) {
            return system.asText();
        }

        // If it's an array of content blocks
        if (system.isArray()) {
            var result = new StringBuilder();
            for (var block : system) {
                if (block.has("text")) {
                    if (result.length() > 0) {
                        result.append("\n\n");
                    }
                    result.append(block.get("text").asText());
                }
            }
            return result.toString();
        }

        return null;
    }
}

                .role("assistant")
                .model(model)
                .content(List.of(ContentBlock.text(text)))
                .stopReason("end_turn")
                .usage(usage)
                .build();
    }

     * Create a response with text and tool uses
     */
    public static AnthropicResponse createWithToolUses(String id, String model, String text, List<ToolUse> toolUses, AnthropicUsage usage) {

                .id(id)
                .type("message")
                .role("assistant")
                .model(model)
                .content(contentBlocks)
                .stopReason(toolUses.isEmpty() ? "end_turn" : "tool_use")
                .usage(usage)
                .build();

import lombok.Builder;

import lombok.extern.jackson.Jacksonized;

/**
 * Anthropic API message
 * Represents a single message in a conversation
 */
@Value
@Builder
@Jacksonized
@JsonInclude(JsonInclude.Include.NON_NULL)
public class Message {

    /**
     * Role: "user" or "assistant"
     */
    String role;

    /**
     * Content can be either:
     * - A simple string
     * - An array of ContentBlock objects
     *
     * Jackson will handle both cases automatically
     */
    JsonNode content;

    /**
     * Create a simple user message with text
     */
    public static Message user(String text) {
        return Message.builder()
                .role("user")
                .content(com.fasterxml.jackson.databind.node.JsonNodeFactory.instance.textNode(text))
                .build();
    }

    /**
     * Create a simple assistant message with text
     */
    public static Message assistant(String text) {
        return Message.builder()
                .role("assistant")
                .content(com.fasterxml.jackson.databind.node.JsonNodeFactory.instance.textNode(text))
                .build();
    }

    /**
     * Extract content from this message
     * Preserves ALL content blocks without filtering
     * - Simple text: returned as-is
     * - Structured content (arrays): serialized as JSON to preserve everything
     */
    public String getTextContent() {
        if (content == null) {
            return "";
        }

        // Simple text - return as-is
        if (content.isTextual()) {
            return content.asText();
        }

        // Structured content (arrays with tool_use, tool_result, etc.)
        // Serialize the entire structure as JSON to preserve everything
        if (content.isArray()) {
            return content.toString();
        }

        // Fallback for other types - serialize as JSON
        return content.toString();
    }
}

/**
 * Anthropic API streaming response chunk
 *
 * Streaming events:
 * - message_start: Initial message metadata
 * - content_block_start: Start of a content block
 * - content_block_delta: Incremental text
 * - content_block_stop: End of a content block
 * - message_delta: Message-level updates (stop_reason, usage)
 * - message_stop: End of message
 * - ping: Keep-alive
 * - error: Error event
 */
@Value
@Builder
@Jacksonized
@JsonInclude(JsonInclude.Include.NON_NULL)
public class StreamingChunk {

    /**
     * Event type
     */
    String type;

    /**
     * Message metadata (for message_start)
     */
    MessageData message;

    /**
     * Content block index (for content_block_start)
     */
    Integer index;

    /**
     * Content block info (for content_block_start)
     */
    @JsonProperty("content_block")
    ContentBlock contentBlock;

    /**
     * Delta update (for content_block_delta)
     */
    Delta delta;

    /**
     * Usage update (for message_delta)
     */
    AnthropicUsage usage;

    /**
     * Error information (for error events)
     */
    ErrorData error;

    /**
     * Message metadata structure
     */
    @Value
    @Builder
    @Jacksonized
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public static class MessageData {
        String id;
        String type;
        String role;
        String model;
        AnthropicUsage usage;

        @JsonProperty("stop_reason")
        String stopReason;

        @JsonProperty("stop_sequence")
        String stopSequence;
    }

    /**
     * Delta update structure
     */
    @Value
    @Builder
    @Jacksonized
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public static class Delta {
        String type;
        String text;

        @JsonProperty("partial_json")
        String partialJson;

        @JsonProperty("stop_reason")
        String stopReason;

        @JsonProperty("stop_sequence")
        String stopSequence;
    }

    /**
     * Error information
     */
    @Value
    @Builder
    @Jacksonized
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public static class ErrorData {
        String type;
        String message;
    }

    /**
     * Create a message_start event
     */
    public static StreamingChunk messageStart(String id, String model) {
        return StreamingChunk.builder()
                .type("message_start")
                .message(MessageData.builder()
                        .id(id)
                        .type("message")
                        .role("assistant")
                        .model(model)
                        .build())
                .build();
    }

    /**
     * Create a content_block_start event
     */
    public static StreamingChunk contentBlockStart(int index) {
        return StreamingChunk.builder()
                .type("content_block_start")
                .index(index)
                .contentBlock(ContentBlock.builder()
                        .type("text")
                        .text("")
                        .build())
                .build();
    }

    /**
     * Create a content_block_delta event with text
     */
    public static StreamingChunk contentBlockDelta(int index, String text) {
        return StreamingChunk.builder()
                .type("content_block_delta")
                .index(index)
                .delta(Delta.builder()
                        .type("text_delta")
                        .text(text)
                        .build())
                .build();
    }

    /**
     * Create a content_block_delta event carrying the tool input as JSON text (input_json_delta)
     */
    public static StreamingChunk inputJsonDelta(int index, String partialJson) {
        return StreamingChunk.builder()
                .type("content_block_delta")
                .index(index)
                .delta(Delta.builder()
                        .type("input_json_delta")
                        .partialJson(partialJson)
                        .build())
                .build();
    }

    /**
     * Create a content_block_start event for tool_use
     */
    public static StreamingChunk toolUseBlockStart(int index, ToolUse toolUse) {
        return StreamingChunk.builder()
                .type("content_block_start")
                .index(index)
                .contentBlock(ContentBlock.toolUse(toolUse))
                .build();
    }

    /**
     * Create a content_block_stop event
     */
    public static StreamingChunk contentBlockStop(int index) {
        return StreamingChunk.builder()
                .type("content_block_stop")
                .index(index)
                .build();
    }

    /**
     * Create a message_delta event with stop reason
     */
    public static StreamingChunk messageDelta(String stopReason, AnthropicUsage usage) {
        return StreamingChunk.builder()
                .type("message_delta")
                .delta(Delta.builder()
                        .stopReason(stopReason)
                        .build())
                .usage(usage)
                .build();
    }

    /**
     * Create a message_stop event
     */
    public static StreamingChunk messageStop() {
        return StreamingChunk.builder()
                .type("message_stop")
                .build();
    }

    /**
     * Create a ping event
     */
    public static StreamingChunk ping() {
        return StreamingChunk.builder()
                .type("ping")
                .build();
    }

    /**
     * Create an error event
     */
    public static StreamingChunk error(String type, String message) {
        return StreamingChunk.builder()
                .type("error")
                .error(ErrorData.builder()
                        .type(type)
                        .message(message)
                        .build())
                .build();
    }

    /**
     * Get the resolved ccapiBaseUrl with environment variable substitution.
     * Supports Windows-style environment variables like %CCAPI_BASE_URL%
     *
     * @return the ccapiBaseUrl with environment variables resolved, or the original value if no variables found
     */
    public String resolvedCcapiBaseUrl() {
        return resolveEnvironmentVariables(ccapiBaseUrl);
    }

    /**
     * Resolve environment variables in a string.
     * Supports Windows-style format: %VARIABLE_NAME%
     *
     * @param value the string that may contain environment variable references
     * @return the string with environment variables substituted, or the original value if null/empty
     */
    private static String resolveEnvironmentVariables(String value) {
        if (value == null || value.isEmpty()) {
            return value;
        }

        // Pattern to match %VARIABLE_NAME%
        var pattern = java.util.regex.Pattern.compile("%([A-Za-z0-9_]+)%");
        var matcher = pattern.matcher(value);
        var result = new StringBuilder();

        var lastEnd = 0;
        while (matcher.find()) {
            // Append text before the match
            result.append(value, lastEnd, matcher.start());

            // Get environment variable name and value
            var varName = matcher.group(1);
            var varValue = System.getenv(varName);

            // Append the resolved value or keep the original if not found
            if (varValue != null) {
                result.append(varValue);
            } else {
                // Keep original if environment variable not found
                result.append(matcher.group(0));
            }

            lastEnd = matcher.end();
        }

        // Append remaining text after last match
        result.append(value.substring(lastEnd));

        return result.toString();
    }
